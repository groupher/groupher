alias GroupherServer.Repo
path = System.fetch_env!("PENDING_AUDIT_OUTPUT")
File.write!(path, "")
:telemetry.attach("pending-audit-query-errors", Repo.config()[:telemetry_prefix] ++ [:query], fn _, _, meta, _ ->
  case meta[:result] do
    {:error, error} ->
      pg = Map.get(error, :postgres, %{})
      File.write!(path, Jason.encode!(%{kind: "query_error", at: DateTime.to_iso8601(DateTime.utc_now()),
        query: meta[:query], error_type: inspect(error.__struct__),
        postgres: Map.take(pg, [:code, :message, :detail, :where, :table, :constraint])}) <> "\n", [:append])
    _ -> nil
  end
end, nil)
Task.start(fn ->
  opts = Keyword.take(Repo.config(), [:hostname, :port, :username, :password, :database])
  {:ok, conn} = Postgrex.start_link(opts)
  loop = fn loop ->
    case Postgrex.query(conn, """
    SELECT waiting.pid, blocker.pid, waiting.wait_event_type, waiting.wait_event,
      blocker.state, waiting.query, blocker.query,
      extract(epoch FROM clock_timestamp()-blocker.xact_start)::float8
    FROM pg_stat_activity waiting
    CROSS JOIN LATERAL unnest(pg_blocking_pids(waiting.pid)) AS held(pid)
    JOIN pg_stat_activity blocker ON blocker.pid=held.pid
    WHERE waiting.datname=current_database()
    """, []) do
      {:ok, %{rows: rows}} when rows != [] ->
        File.write!(path, Jason.encode!(%{kind: "blocking", rows: rows}) <> "\n", [:append])
      _ -> nil
    end
    Process.sleep(100)
    loop.(loop)
  end
  loop.(loop)
end)
