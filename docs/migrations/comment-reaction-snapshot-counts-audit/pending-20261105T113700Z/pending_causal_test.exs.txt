defmodule PendingCausalAuditTest do
  use ExUnit.Case, async: false
  import Ecto.Query
  import GroupherServer.Support.Factory
  alias Ecto.Adapters.SQL.Sandbox
  alias GroupherServer.{CMS, Repo}
  alias CMS.Model.{Article, ArticleStats, Comment}
  alias CMS.ArtimentMentions.MentionsGraphRevision, as: Revision

  defp fixture do
    Sandbox.unboxed_run(Repo, fn ->
      {:ok, actor} = db_insert(:user)
      {:ok, community} = mock_community(actor)
      attrs = mock_attrs(:post, %{community_id: community.id, author: %{user: actor}})
      {community, actor, Map.put(attrs, :body_bag, mock_body_bag(attrs.body))}
    end)
  end

  test "Sandbox outer transaction retains the global row lock after domain transaction returns" do
    {community, actor, attrs} = fixture()
    parent = self()
    command = Ecto.UUID.generate()
    holder = Task.async(fn ->
      :ok = Sandbox.checkout(Repo)
      {:ok, :pass} = Repo.transact_result(fn -> Revision.bump() end)
      send(parent, :sandbox_holder_ready)
      receive do :release -> :ok after 15_000 -> raise "holder barrier timeout" end
      :ok = Sandbox.checkin(Repo)
    end)
    assert_receive :sandbox_holder_ready, 5_000
    contender = Task.async(fn ->
      :ok = Sandbox.checkout(Repo)
      start = System.monotonic_time(:millisecond)
      first = CMS.Articles.create(community, :post, attrs, actor, command_id: command)
      elapsed = System.monotonic_time(:millisecond) - start
      send(parent, {:first, first, elapsed})
      receive do :retry -> :ok after 15_000 -> raise "retry barrier timeout" end
      second = CMS.Articles.create(community, :post, attrs, actor, command_id: command)
      {:ok, article} = second
      assert Repo.aggregate(from(a in Article, where: a.id == ^article.article_id), :count) == 1
      stats = Repo.get_by!(ArticleStats, article_id: article.article_id)
      :ok = Sandbox.checkin(Repo)
      %{first_reason: elem(first,1).reason, elapsed_ms: elapsed, retry: "ok", comments_count: stats.comments_count, comments_revision: stats.comments_revision}
    end)
    assert_receive {:first, {:error, %{reason: :command_resolution_pending}}, elapsed}, 10_000
    assert elapsed >= 3900
    send(holder.pid, :release)
    Task.await(holder, 5_000)
    send(contender.pid, :retry)
    result = Task.await(contender, 10_000)
    File.write!("/tmp/pending-sandbox-causal.json", Jason.encode!(result, pretty: true))
    File.write!("/tmp/pending-causal-repeats.jsonl", Jason.encode!(Map.put(result, :kind, "sandbox")) <> "\n", [:append])
  end

  test "independent committed business commands succeed and preserve comment facts and summaries" do
    fixtures = Enum.map(1..4, fn _ -> fixture() end)
    parent = self()
    burst = fn entries, fun ->
      tasks = Enum.map(entries, fn entry ->
        Task.async(fn -> Sandbox.unboxed_run(Repo, fn ->
          send(parent, {:ready, self()})
          receive do :go -> fun.(entry) after 10_000 -> raise "business barrier timeout" end
        end) end)
      end)
      for _ <- tasks, do: assert_receive({:ready, _}, 10_000)
      start = System.monotonic_time(:millisecond)
      Enum.each(tasks, &send(&1.pid, :go))
      results = Enum.map(tasks, &Task.await(&1, 30_000))
      {results, System.monotonic_time(:millisecond)-start}
    end
    {created, create_ms} = burst.(fixtures, fn {community, actor, attrs} ->
      CMS.Articles.create(community, :post, attrs, actor, command_id: Ecto.UUID.generate())
    end)
    assert Enum.all?(created, &match?({:ok, _}, &1))
    {:ok, article} = hd(created)
    {community, _, _} = hd(fixtures)
    {comments, comment_ms} = burst.(fixtures, fn {_, actor, _} ->
      CMS.Comments.create_comment(community, :post, article.inner_id, mock_comment(), actor, Ecto.UUID.generate())
    end)
    assert Enum.all?(comments, &match?({:ok, _}, &1))
    stats = Sandbox.unboxed_run(Repo, fn ->
      assert Repo.aggregate(from(c in Comment, where: c.article_id == ^article.article_id), :count) == 4
      Repo.get_by!(ArticleStats, article_id: article.article_id)
    end)
    assert stats.comments_count == 4
    assert stats.comments_participants_count == 4
    assert stats.comments_revision == 5
    assert length(stats.comments_participants) == 4
    business_result = %{create_successes: 4, comment_successes: 4,
      create_burst_ms: create_ms, comment_burst_ms: comment_ms, comments_count: stats.comments_count,
      participants_count: stats.comments_participants_count, comments_revision: stats.comments_revision}
    File.write!("/tmp/pending-business-causal.json", Jason.encode!(business_result, pretty: true))
    File.write!("/tmp/pending-causal-repeats.jsonl", Jason.encode!(Map.put(business_result, :kind, "committed_business")) <> "\n", [:append])
  end
end
