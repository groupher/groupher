defmodule GroupherServer.CMS.ArtimentMentions.Store do
  @moduledoc """
  Stores stable Mention facts and synchronizes each canonical public source.

      canonical BodyBag -> parsed targets -> locked fact replacement + synced token
  """

  import Ecto.Query, warn: false

  alias GroupherServer.{CMS, Repo}

  alias CMS.ArtimentMentions.MentionsGraphRevision, as: Revision

  alias CMS.{
    ArtimentMentions.Model.CurrentEdge,
    ArtimentMentions.Model.Endpoint,
    ArtimentMentions.Model.MentionSet,
    ArtimentMentions.Model.Occurrence,
    ArtimentMentions.Model.ProjectionState
  }

  alias CMS.Model.Comment

  @type target_state :: :active | :trashed | :permanently_deleted
  @type sync_output :: :pass | {non_neg_integer(), nil}
  @type sync_result :: {:ok, sync_output()} | {:error, term()}
  @version_fact_fields [
    :occurrence_key,
    :mention_case,
    :target_type,
    :target_endpoint_id,
    :target_user_id,
    :normalized_url,
    :url_hash,
    :editor_target_id,
    :block_id,
    :path,
    :mentioned_at
  ]

  @doc "Creates or reuses one stable endpoint after its canonical identity was verified."
  @spec ensure_endpoint_in_transaction(map()) :: {:ok, struct()} | {:error, term()}
  def ensure_endpoint_in_transaction(attrs) when is_map(attrs) do
    with {:ok, identity} <- endpoint_identity(attrs),
         {:ok, :pass} <- validate_parent_endpoint(identity) do
      endpoint_query(identity)
      |> Repo.one()
      |> case do
        %CMS.ArtimentMentions.Model.Endpoint{} = endpoint ->
          if endpoint_identity(endpoint) == {:ok, identity} do
            {:ok, endpoint}
          else
            {:error, :mention_endpoint_identity_conflict}
          end

        nil ->
          insert_or_get_endpoint(identity)
      end
    end
  end

  def ensure_endpoint_in_transaction(_), do: {:error, :invalid_mention_endpoint}

  @doc "Reuses prior internal bindings for unresolved or relinked parser candidates."
  def inherit_unresolved_candidates(source_identity, candidates, resolved_mentions)
      when is_map(source_identity) and is_list(candidates) and is_list(resolved_mentions) do
    with {:ok, source_endpoint_id} <- ensure_source_endpoint(source_identity) do
      Repo.get!(Endpoint, source_endpoint_id, lock: "FOR UPDATE")

      previous_set_id =
        case Repo.get(ProjectionState, source_endpoint_id) do
          %ProjectionState{desired_set_id: set_id} -> set_id
          _ -> nil
        end

      case previous_set_id do
        nil ->
          {:ok, resolved_mentions}

        set_id ->
          previous_occurrences =
            from(occurrence in Occurrence,
              where: occurrence.set_id == ^set_id,
              where: not is_nil(occurrence.target_endpoint_id)
            )
            |> Repo.all()

          endpoint_ids = previous_occurrences |> Enum.map(& &1.target_endpoint_id) |> Enum.uniq()

          endpoints =
            from(endpoint in Endpoint, where: endpoint.id in ^endpoint_ids)
            |> Repo.all()
            |> Map.new(&{&1.id, &1})

          inherit_candidates(candidates, resolved_mentions, previous_occurrences, endpoints)
      end
    end
  end

  def inherit_unresolved_candidates(_, _, _) do
    {:error, :invalid_mention_inheritance}
  end

  @doc "Hides one source's outgoing current edges while preserving its endpoint and facts."
  def hide_source(source_identity) when is_map(source_identity) do
    transaction(fn -> hide_source_in_transaction(source_identity) end)
  end

  def hide_source(_), do: {:error, :invalid_mention_endpoint}

  @doc "Marks a permanently removed target and removes only its outgoing current edges."
  def mark_endpoint_deleted_in_transaction(source_identity) when is_map(source_identity) do
    with {:ok, identity} <- endpoint_identity(source_identity) do
      case endpoint_query(identity) |> lock("FOR UPDATE") |> Repo.one() do
        %Endpoint{deleted_at: %DateTime{}} = endpoint ->
          with {:ok, removed_count} <- remove_outgoing_edges(endpoint),
               {:ok, child_count} <- remove_comment_child_edges(endpoint),
               {:ok, _cleared} <- discard_source_facts(endpoint.id) do
            if removed_count > 0 or child_count > 0 do
              Revision.bump()
            else
              {:ok, :pass}
            end
          end

        %Endpoint{} = endpoint ->
          now = DateTime.utc_now() |> DateTime.truncate(:second)

          with {:ok, _endpoint} <-
                 endpoint
                 |> Endpoint.changeset(%{deleted_at: now})
                 |> Repo.update(),
               {:ok, _removed_count} <- remove_outgoing_edges(endpoint) do
            with {:ok, _child_count} <- remove_comment_child_edges(endpoint),
                 {:ok, _cleared} <- discard_source_facts(endpoint.id) do
              bump_endpoint_revision(endpoint)
            end
          end

        nil ->
          {:ok, :pass}
      end
    end
  end

  def mark_endpoint_deleted_in_transaction(_), do: {:error, :invalid_mention_endpoint}

  defp hide_source_in_transaction(source_identity) do
    with {:ok, identity} <- endpoint_identity(source_identity) do
      case endpoint_query(identity) |> lock("FOR UPDATE") |> Repo.one() do
        %Endpoint{} = endpoint ->
          with {:ok, removed_count} <- remove_outgoing_edges(endpoint) do
            if removed_count > 0, do: bump_endpoint_revision(endpoint), else: {:ok, :pass}
          end

        nil ->
          {:ok, :pass}
      end
    end
  end

  defp remove_outgoing_edges(%Endpoint{id: endpoint_id}) do
    {count, nil} =
      Repo.delete_all(from(edge in CurrentEdge, where: edge.source_endpoint_id == ^endpoint_id))

    {:ok, count}
  end

  defp remove_comment_child_edges(%Endpoint{id: parent_endpoint_id}) do
    child_ids =
      from(endpoint in Endpoint,
        where: endpoint.type == :comment and endpoint.parent_endpoint_id == ^parent_endpoint_id,
        order_by: endpoint.id,
        lock: "FOR UPDATE",
        select: endpoint.id
      )
      |> Repo.all()

    if child_ids == [] do
      {:ok, 0}
    else
      {removed_count, nil} =
        Repo.delete_all(from(edge in CurrentEdge, where: edge.source_endpoint_id in ^child_ids))

      Enum.each(child_ids, fn id ->
        {:ok, _} = discard_source_facts(id)
      end)

      {:ok, removed_count}
    end
  end

  defp discard_source_facts(id) do
    Repo.delete_all(from(state in ProjectionState, where: state.source_endpoint_id == ^id))
    {count, nil} = Repo.delete_all(from(set in MentionSet, where: set.source_endpoint_id == ^id))
    {:ok, count}
  end

  defp bump_endpoint_revision(%Endpoint{}), do: Revision.bump()

  defp transaction(fun) when is_function(fun, 0) do
    if Repo.in_transaction?() do
      fun.()
    else
      case Repo.transaction(fun) do
        {:ok, result} -> result
        {:error, reason} -> {:error, reason}
      end
    end
  end

  @doc "Resolves parsed targets and saves one immutable Mention version set."
  def save_parsed_version(
        source_identity,
        source_token,
        parser_version,
        mentions,
        mentioned_at,
        opts \\ []
      )

  def save_parsed_version(
        source_identity,
        source_token,
        parser_version,
        mentions,
        mentioned_at,
        opts
      )
      when is_map(source_identity) and is_binary(source_token) and is_list(mentions) and
             is_integer(parser_version) and parser_version > 0 and
             is_struct(mentioned_at, DateTime) and is_list(opts) do
    Repo.transaction(fn ->
      case save_parsed_version_in_transaction(
             source_identity,
             source_token,
             parser_version,
             mentions,
             mentioned_at,
             opts
           ) do
        {:ok, %{set: set}} -> set
        {:error, reason} -> Repo.rollback(reason)
      end
    end)
    |> version_fact_result()
  end

  def save_parsed_version(_, _, _, _, _, _), do: {:error, :invalid_mention_set}

  @doc "Saves parsed facts and desired watermark inside the caller's source transaction."
  def save_parsed_version_in_transaction(
        source_identity,
        source_token,
        parser_version,
        mentions,
        mentioned_at,
        opts \\ []
      )

  def save_parsed_version_in_transaction(
        source_identity,
        source_token,
        parser_version,
        mentions,
        mentioned_at,
        opts
      )
      when is_map(source_identity) and is_binary(source_token) and is_list(mentions) and
             is_integer(parser_version) and parser_version > 0 and
             is_struct(mentioned_at, DateTime) and is_list(opts) do
    with {:ok, source_endpoint_id} <- ensure_source_endpoint(source_identity),
         %Endpoint{} <- Repo.get(Endpoint, source_endpoint_id, lock: "FOR UPDATE"),
         previous_user_ids <- previous_user_ids(source_endpoint_id),
         occurrences <- parsed_occurrences(mentions, mentioned_at, source_endpoint_id, opts),
         {:ok, set} <-
           save_version_facts(source_endpoint_id, source_token, parser_version, occurrences),
         {:ok, _state} <- mark_desired_set(source_endpoint_id, source_token, set) do
      user_ids =
        occurrences |> Enum.filter(&(&1.target_type == :user)) |> Enum.map(& &1.target_user_id)

      new_user_ids = Enum.uniq(user_ids) -- previous_user_ids

      {:ok,
       %{
         source_endpoint_id: source_endpoint_id,
         set: set,
         new_user_ids: new_user_ids
       }}
    end
  end

  def save_parsed_version_in_transaction(_, _, _, _, _, _) do
    {:error, :invalid_mention_set}
  end

  defp previous_user_ids(source_id) do
    case Repo.get(ProjectionState, source_id) do
      %ProjectionState{desired_set_id: set_id} ->
        from(o in Occurrence,
          where: o.set_id == ^set_id and o.target_type == :user,
          select: o.target_user_id,
          distinct: true
        )
        |> Repo.all()

      _ ->
        []
    end
  end

  defp parsed_occurrences(mentions, mentioned_at, source_endpoint_id, opts) do
    previous_set_id =
      Keyword.get_lazy(opts, :previous_set_id, fn ->
        case Repo.get(ProjectionState, source_endpoint_id) do
          %ProjectionState{desired_set_id: set_id} -> set_id
          _ -> nil
        end
      end)

    mentions
    |> Enum.map(fn mention ->
      case occurrence_attrs(mention, mentioned_at) do
        {:ok, attrs} -> attrs
        {:error, reason} -> Repo.rollback(reason)
      end
    end)
    |> inherit_mentioned_at(source_endpoint_id, previous_set_id)
  end

  @doc "Projects one complete desired set and watermark atomically."
  @spec project_set(Ecto.UUID.t(), Ecto.UUID.t()) :: {:ok, non_neg_integer()} | {:error, term()}
  def project_set(source_endpoint_id, set_id, opts \\ [])

  def project_set(source_endpoint_id, set_id, opts)
      when is_binary(source_endpoint_id) and is_binary(set_id) and is_list(opts) do
    case Repo.transaction(fn ->
           case project_set_in_transaction(
                  source_endpoint_id,
                  set_id,
                  false,
                  Keyword.get(opts, :verify_current, false)
                ) do
             {:ok, count} -> count
             {:error, reason} -> Repo.rollback(reason)
           end
         end) do
      {:ok, count} -> {:ok, count}
      {:error, reason} -> {:error, reason}
    end
  end

  def project_set(_, _, _), do: {:error, :invalid_mention_projection}

  @doc "Rebuilds the canonical current index from its saved set without parsing or notifying."
  def rebuild_current_set(source_endpoint_id) when is_binary(source_endpoint_id) do
    Repo.transaction(fn ->
      with %Endpoint{deleted_at: nil} = source <-
             Repo.get(Endpoint, source_endpoint_id, lock: "FOR UPDATE"),
           %ProjectionState{} = state <-
             Repo.get(ProjectionState, source_endpoint_id, lock: "FOR UPDATE"),
           {:ok, selected} <- CMS.ArtimentMentions.CurrentSource.load(source),
           true <- state.desired_content_token == selected.token,
           %MentionSet{source_token: token} <- Repo.get(MentionSet, state.desired_set_id),
           true <- token == selected.token,
           {:ok, count} <-
             project_set_in_transaction(source_endpoint_id, state.desired_set_id, true) do
        count
      else
        %Endpoint{} -> Repo.rollback(:mention_projection_source_deleted)
        nil -> Repo.rollback(:mention_current_set_missing)
        false -> Repo.rollback(:mention_current_set_stale)
        {:error, reason} -> Repo.rollback(reason)
      end
    end)
  end

  def rebuild_current_set(_), do: {:error, :invalid_mention_projection}

  defp project_set_in_transaction(source_endpoint_id, set_id, force, verify_current \\ false) do
    source =
      from(endpoint in Endpoint,
        where: endpoint.id == ^source_endpoint_id,
        lock: "FOR UPDATE"
      )
      |> Repo.one()

    with %Endpoint{deleted_at: nil} <- source,
         %ProjectionState{} = state <-
           Repo.get(ProjectionState, source_endpoint_id, lock: "FOR UPDATE"),
         true <- state.desired_set_id == set_id,
         %MentionSet{source_endpoint_id: ^source_endpoint_id} = set <-
           Repo.get(MentionSet, set_id),
         true <- state.desired_content_token == set.source_token,
         {:ok, :pass} <- verify_current_selection(source, set, verify_current) do
      if not force and state.synced_set_id == set.id and
           state.synced_content_token == set.source_token do
        {:ok, 0}
      else
        occurrences =
          from(occurrence in Occurrence,
            where: occurrence.set_id == ^set.id,
            order_by: [asc: occurrence.occurrence_key]
          )
          |> Repo.all()

        occurrence_attrs = Enum.map(occurrences, &Map.from_struct/1)

        with {:ok, :pass} <- validate_occurrence_times(occurrence_attrs),
             {:ok, grouped_edges} <- grouped_current_edges(source_endpoint_id, set, occurrences) do
          if force and state.synced_set_id == set.id and
               current_index_matches?(source.id, grouped_edges) do
            {:ok, 0}
          else
            with {_, nil} <-
                   Repo.delete_all(
                     from(edge in CurrentEdge, where: edge.source_endpoint_id == ^source.id)
                   ),
                 {:ok, count} <- insert_current_edges(grouped_edges),
                 {:ok, _state} <- maybe_mark_projected(state, set),
                 {:ok, _collected} <- collect_old_comment_sets(source, set.id),
                 {:ok, :pass} <- bump_projected_source_revision(source) do
              {:ok, count}
            end
          end
        end
      end
    else
      %Endpoint{} -> {:error, :mention_projection_source_deleted}
      nil -> {:error, :mention_projection_source_missing}
      false -> {:error, :mention_projection_set_not_desired}
      %MentionSet{} -> {:error, :mention_projection_set_source_mismatch}
      {:error, _reason} = error -> error
      _ -> {:error, :mention_projection_state_missing}
    end
  end

  defp verify_current_selection(_source, _set, false), do: {:ok, :pass}

  defp verify_current_selection(source, set, true) do
    case CMS.ArtimentMentions.CurrentSource.load(source) do
      {:ok, %{token: token}} when token == set.source_token -> {:ok, :pass}
      {:ok, _} -> {:error, :mention_projection_set_not_desired}
      {:error, :mention_current_source_missing} -> {:error, :mention_projection_source_unselected}
      {:error, reason} -> {:error, reason}
    end
  end

  defp collect_old_comment_sets(%Endpoint{type: :comment, id: id}, current_set_id) do
    {count, nil} =
      Repo.delete_all(
        from(set in MentionSet,
          where: set.source_endpoint_id == ^id and set.id != ^current_set_id
        )
      )

    {:ok, count}
  end

  defp collect_old_comment_sets(_source, _current_set_id), do: {:ok, 0}

  defp current_index_matches?(source_id, expected) do
    actual = from(edge in CurrentEdge, where: edge.source_endpoint_id == ^source_id) |> Repo.all()

    fields = [
      :source_endpoint_id,
      :source_set_id,
      :target_type,
      :target_endpoint_id,
      :target_user_id,
      :normalized_url,
      :url_hash,
      :mention_case,
      :edge_key,
      :mentioned_at,
      :occurrences
    ]

    normalize = fn edges ->
      edges
      |> Enum.map(&Map.take(&1, fields))
      |> Enum.sort_by(& &1.edge_key)
      |> Jason.encode!()
      |> Jason.decode!()
    end

    normalize.(actual) == normalize.(expected)
  end

  defp maybe_mark_projected(
         %ProjectionState{synced_set_id: id, synced_content_token: token} = state,
         %MentionSet{id: id, source_token: token}
       ) do
    {:ok, state}
  end

  defp maybe_mark_projected(state, set), do: mark_projected(state, set)

  defp grouped_current_edges(source_endpoint_id, set, occurrences) do
    grouped = Enum.group_by(occurrences, &occurrence_edge_identity/1)

    edges =
      Enum.map(grouped, fn {identity, facts} ->
        [first | _] = facts

        %{
          source_endpoint_id: source_endpoint_id,
          source_set_id: set.id,
          target_type: first.target_type,
          target_endpoint_id: first.target_endpoint_id,
          target_user_id: first.target_user_id,
          normalized_url: current_target_url(first),
          url_hash: current_target_url_hash(first),
          mention_case: first.mention_case,
          edge_key: current_edge_key(source_endpoint_id, identity),
          mentioned_at: first.mentioned_at,
          occurrences: %{items: Enum.map(facts, &current_occurrence/1)}
        }
      end)
      |> Enum.sort_by(& &1.edge_key)

    {:ok, edges}
  end

  defp occurrence_edge_identity(occurrence) do
    identity =
      Map.new(
        [:mention_case, :target_type, :target_endpoint_id, :target_user_id, :normalized_url],
        &{&1, Map.get(occurrence, &1)}
      )

    if occurrence.target_type == :url do
      identity
    else
      Map.put(identity, :normalized_url, nil)
    end
  end

  defp current_target_url(%Occurrence{target_type: :url, normalized_url: url}), do: url
  defp current_target_url(_occurrence), do: nil

  defp current_target_url_hash(%Occurrence{target_type: :url, url_hash: hash}), do: hash
  defp current_target_url_hash(_occurrence), do: nil

  defp current_edge_key(source_endpoint_id, identity) do
    :crypto.hash(:sha256, :erlang.term_to_binary({source_endpoint_id, identity}))
    |> Base.encode16(case: :lower)
  end

  defp current_occurrence(occurrence) do
    %{
      occurrence_key: occurrence.occurrence_key,
      editor_target_id: occurrence.editor_target_id,
      block_id: occurrence.block_id,
      path: occurrence.path
    }
  end

  defp insert_current_edges(edges) do
    Enum.reduce_while(edges, {:ok, 0}, fn attrs, {:ok, count} ->
      case Repo.insert(CurrentEdge.changeset(%CurrentEdge{}, attrs)) do
        {:ok, _edge} -> {:cont, {:ok, count + 1}}
        {:error, changeset} -> {:halt, {:error, changeset}}
      end
    end)
  end

  defp mark_projected(state, set) do
    state
    |> ProjectionState.changeset(%{
      synced_content_token: set.source_token,
      synced_set_id: set.id,
      synced_at: DateTime.utc_now() |> DateTime.truncate(:second)
    })
    |> Repo.update()
  end

  defp bump_projected_source_revision(%{type: :comment} = source) do
    case Repo.get(Endpoint, source.parent_endpoint_id) do
      %{article_id: article_id} when is_binary(article_id) ->
        Revision.bump()

      _ ->
        {:error, :mention_projection_parent_missing}
    end
  end

  defp bump_projected_source_revision(%{article_id: article_id}) when is_binary(article_id) do
    Revision.bump()
  end

  defp bump_projected_source_revision(_), do: {:error, :invalid_mention_projection_source}

  defp inherit_mentioned_at(occurrences, _source_endpoint_id, nil), do: occurrences

  defp inherit_mentioned_at(occurrences, source_endpoint_id, previous_set_id)
       when is_binary(previous_set_id) do
    previous_set = Repo.get(CMS.ArtimentMentions.Model.MentionSet, previous_set_id)

    unless previous_set && previous_set.source_endpoint_id == source_endpoint_id do
      Repo.rollback(:invalid_previous_mention_set)
    end

    previous_occurrences =
      from(occurrence in CMS.ArtimentMentions.Model.Occurrence,
        where: occurrence.set_id == ^previous_set_id
      )
      |> Repo.all()

    timestamps_by_identity =
      Map.new(previous_occurrences, &{occurrence_edge_identity(&1), &1.mentioned_at})

    Enum.map(occurrences, fn occurrence ->
      case Map.fetch(timestamps_by_identity, occurrence_edge_identity(occurrence)) do
        {:ok, mentioned_at} -> Map.put(occurrence, :mentioned_at, mentioned_at)
        :error -> occurrence
      end
    end)
  end

  defp inherit_mentioned_at(_occurrences, _source_endpoint_id, _previous_set_id) do
    Repo.rollback(:invalid_previous_mention_set)
  end

  defp mark_desired_set(source_endpoint_id, source_token, set) do
    alias CMS.ArtimentMentions.Model.ProjectionState

    case Repo.get(ProjectionState, source_endpoint_id, lock: "FOR UPDATE") do
      nil ->
        with {:ok, state} <-
               Repo.insert(
                 ProjectionState.changeset(%ProjectionState{}, %{
                   source_endpoint_id: source_endpoint_id,
                   desired_content_token: source_token,
                   desired_set_id: set.id
                 })
               ),
             {:ok, :pass} <- Revision.bump() do
          {:ok, state}
        end

      %ProjectionState{} = state ->
        if state.desired_set_id == set.id and state.desired_content_token == source_token do
          {:ok, state}
        else
          with {:ok, updated} <-
                 state
                 |> ProjectionState.changeset(%{
                   desired_content_token: source_token,
                   desired_set_id: set.id
                 })
                 |> Repo.update(),
               {:ok, :pass} <- Revision.bump() do
            {:ok, updated}
          end
        end
    end
  end

  defp ensure_source_endpoint(%{type: :comment, parent: parent} = identity) do
    with {:ok, parent_endpoint} <- ensure_endpoint_in_transaction(parent),
         {:ok, endpoint} <-
           ensure_endpoint_in_transaction(
             identity
             |> Map.delete(:parent)
             |> Map.put(:parent_endpoint_id, parent_endpoint.id)
           ) do
      {:ok, endpoint.id}
    end
  end

  defp ensure_source_endpoint(identity) do
    with {:ok, endpoint} <- ensure_endpoint_in_transaction(identity), do: {:ok, endpoint.id}
  end

  defp occurrence_attrs(
         %{mentioned_scope: :external, mentioned_type: :url} = mention,
         mentioned_at
       ) do
    url = Map.get(mention, :mentioned_url)
    hash = Map.get(mention, :mentioned_url_hash)

    if is_binary(url) and is_binary(hash) do
      {:ok,
       base_occurrence(mention, mentioned_at)
       |> Map.merge(%{
         target_type: :url,
         normalized_url: url,
         url_hash: hash
       })}
    else
      {:error, :invalid_mention_url_target}
    end
  end

  defp occurrence_attrs(
         %{mentioned_scope: :internal, mentioned_type: :user} = mention,
         mentioned_at
       ) do
    case Map.get(mention, :mentioned_id) do
      id when is_integer(id) ->
        {:ok,
         Map.merge(base_occurrence(mention, mentioned_at), %{
           target_type: :user,
           target_user_id: id
         })}

      _ ->
        {:error, :invalid_mention_user_target}
    end
  end

  defp occurrence_attrs(
         %{mentioned_scope: :internal, mentioned_type: type} = mention,
         mentioned_at
       )
       when type in [:post, :blog, :changelog, :doc, :ticket, :comment] and
              not is_map_key(mention, :trusted_target_endpoint_id) do
    with {:ok, endpoint} <- target_endpoint(mention, type) do
      {:ok,
       Map.merge(base_occurrence(mention, mentioned_at), %{
         target_type: type,
         target_endpoint_id: endpoint.id,
         normalized_url: inherited_url(mention)
       })}
    end
  end

  defp occurrence_attrs(
         %{
           mentioned_scope: :internal,
           mentioned_type: type,
           trusted_target_endpoint_id: endpoint_id
         } = mention,
         mentioned_at
       )
       when type in [:post, :blog, :changelog, :doc, :ticket, :comment] and
              is_binary(endpoint_id) do
    case Repo.get_by(Endpoint, id: endpoint_id, type: type) do
      %Endpoint{} ->
        {:ok,
         Map.merge(base_occurrence(mention, mentioned_at), %{
           target_type: type,
           target_endpoint_id: endpoint_id,
           normalized_url: inherited_url(mention)
         })}

      nil ->
        {:error, :invalid_inherited_mention_target}
    end
  end

  defp occurrence_attrs(_, _), do: {:error, :invalid_mention_target}

  defp inherit_candidates(candidates, resolved_mentions, previous_occurrences, endpoints) do
    Enum.reduce_while(candidates, {:ok, %{}}, fn
      %{internal_mention_candidate?: true, type: type} = candidate, {:ok, inherited} ->
        matches =
          previous_occurrences
          |> Enum.filter(fn occurrence ->
            endpoint = Map.get(endpoints, occurrence.target_endpoint_id)

            endpoint && endpoint.type == type &&
              candidate_matches_endpoint?(candidate, occurrence, endpoint)
          end)
          |> Enum.map(& &1.target_endpoint_id)
          |> Enum.uniq()

        case matches do
          [endpoint_id] ->
            key = candidate_occurrence_key(candidate.occurrence)

            mention =
              inherited_mention(candidate, Map.fetch!(endpoints, endpoint_id), endpoint_id)

            {:cont, {:ok, Map.put(inherited, key, mention)}}

          [] ->
            {:cont, {:ok, inherited}}

          _ ->
            {:halt, {:error, :ambiguous_mention_inheritance}}
        end

      _, acc ->
        {:cont, acc}
    end)
    |> case do
      {:ok, inherited} ->
        mentions =
          Enum.reject(resolved_mentions, fn mention ->
            Map.has_key?(inherited, candidate_occurrence_key(Map.get(mention, :occurrence, %{})))
          end)

        {:ok, mentions ++ Map.values(inherited)}

      {:error, _reason} = error ->
        error
    end
  end

  defp candidate_matches_endpoint?(candidate, occurrence, endpoint) do
    raw_url = get_in(candidate, [:occurrence, :raw_url])
    editor_target_id = get_in(candidate, [:occurrence, :editor_target_id])

    cond do
      is_binary(raw_url) and is_binary(occurrence.normalized_url) ->
        normalize_mention_url(raw_url) == occurrence.normalized_url

      is_binary(editor_target_id) and editor_target_id == occurrence.editor_target_id ->
        true

      candidate.type == :comment ->
        cast_endpoint_integer(candidate.value) == endpoint.comment_id

      candidate.type == :doc ->
        case cast_endpoint_uuid(candidate.value) do
          {:ok, article_id} ->
            endpoint.article_id == article_id

          :error ->
            false
        end

      candidate.type in [:post, :blog, :changelog, :ticket] ->
        case cast_endpoint_uuid(candidate.value) do
          {:ok, article_id} -> endpoint.article_id == article_id
          :error -> false
        end

      true ->
        false
    end
  end

  defp inherited_mention(candidate, endpoint, endpoint_id) do
    %{
      mentioned_scope: :internal,
      mentioned_type: endpoint.type,
      mentioned_id: endpoint.article_id || endpoint.comment_id,
      mention_case:
        if(get_in(candidate, [:occurrence, :normalized_from]) == "link",
          do: :link,
          else: :inline_mention
        ),
      trusted_target_endpoint_id: endpoint_id,
      occurrence: candidate.occurrence
    }
  end

  defp candidate_occurrence_key(occurrence) do
    "#{Map.get(occurrence, :block_id) || "block"}:#{Enum.join(Map.get(occurrence, :path, []), ".")}"
  end

  defp cast_endpoint_integer(value) when is_integer(value), do: value

  defp cast_endpoint_integer(value) when is_binary(value) do
    case Integer.parse(value) do
      {id, ""} -> id
      _ -> nil
    end
  end

  defp cast_endpoint_integer(_), do: nil

  defp cast_endpoint_uuid(value) when is_binary(value) do
    case Ecto.UUID.cast(value) do
      {:ok, id} -> {:ok, id}
      :error -> :error
    end
  end

  defp cast_endpoint_uuid(_), do: :error

  defp inherited_url(mention) do
    case get_in(mention, [:occurrence, :raw_url]) do
      raw_url when is_binary(raw_url) -> normalize_mention_url(raw_url)
      _ -> nil
    end
  end

  defp normalize_mention_url(url) do
    uri = URI.parse(String.trim(url))
    scheme = if is_binary(uri.scheme), do: String.downcase(uri.scheme), else: nil
    host = if is_binary(uri.host), do: String.downcase(uri.host), else: nil

    port =
      case {scheme, uri.port} do
        {"http", 80} -> nil
        {"https", 443} -> nil
        {_, value} -> value
      end

    uri
    |> Map.put(:scheme, scheme)
    |> Map.put(:host, host)
    |> Map.put(:port, port)
    |> URI.to_string()
  end

  defp target_endpoint(%{artiment: %{article_id: article_id}}, type)
       when type in [:post, :blog, :changelog, :doc, :ticket] and is_binary(article_id) do
    ensure_endpoint_in_transaction(%{
      type: type,
      article_id: article_id
    })
  end

  defp target_endpoint(%{artiment: %Comment{} = comment, parent_article: parent}, :comment)
       when is_map(parent) do
    parent_identity = %{
      type: Map.get(parent, :thread),
      article_id: Map.get(parent, :article_id)
    }

    with {:ok, parent_endpoint} <- ensure_endpoint_in_transaction(parent_identity) do
      ensure_endpoint_in_transaction(%{
        type: :comment,
        comment_id: comment.id,
        parent_endpoint_id: parent_endpoint.id
      })
    end
  end

  defp target_endpoint(_, _), do: {:error, :invalid_mention_target_identity}

  defp base_occurrence(mention, mentioned_at) do
    position = Map.get(mention, :occurrence, %{})
    path = Map.get(position, :path, [])
    block_id = Map.get(position, :block_id)

    %{
      occurrence_key: "#{block_id || "block"}:#{Enum.join(path, ".")}",
      mention_case: Map.get(mention, :mention_case),
      editor_target_id: Map.get(position, :editor_target_id),
      block_id: block_id,
      path: Enum.join(path, "."),
      mentioned_at: mentioned_at
    }
  end

  defp endpoint_identity(%{type: type, article_id: article_id} = attrs)
       when type in [:post, :blog, :changelog, :doc, :ticket] and is_binary(article_id) do
    if Map.get(attrs, :branch_id) || Map.get(attrs, :comment_id) ||
         Map.get(attrs, :parent_endpoint_id) do
      {:error, :invalid_mention_endpoint}
    else
      {:ok, %{type: type, article_id: article_id, comment_id: nil}}
    end
  end

  defp endpoint_identity(
         %{type: :comment, comment_id: comment_id, parent_endpoint_id: parent_id} = attrs
       )
       when is_integer(comment_id) and is_binary(parent_id) do
    if Map.get(attrs, :article_id) || Map.get(attrs, :branch_id) do
      {:error, :invalid_mention_endpoint}
    else
      {:ok, Map.take(attrs, [:type, :comment_id, :parent_endpoint_id])}
    end
  end

  defp endpoint_identity(_), do: {:error, :invalid_mention_endpoint}

  defp validate_parent_endpoint(%{type: :comment} = identity) do
    case Repo.get(CMS.ArtimentMentions.Model.Endpoint, identity.parent_endpoint_id) do
      %CMS.ArtimentMentions.Model.Endpoint{type: type}
      when type in [:post, :blog, :changelog, :doc] ->
        {:ok, :pass}

      _ ->
        {:error, :mention_parent_endpoint_not_found}
    end
  end

  defp validate_parent_endpoint(_identity), do: {:ok, :pass}

  defp endpoint_query(%{type: :comment, comment_id: comment_id}) do
    from(endpoint in CMS.ArtimentMentions.Model.Endpoint,
      where: endpoint.type == :comment and endpoint.comment_id == ^comment_id
    )
  end

  defp endpoint_query(%{type: type, article_id: article_id}) do
    from(endpoint in CMS.ArtimentMentions.Model.Endpoint,
      where: endpoint.type == ^type and endpoint.article_id == ^article_id
    )
  end

  defp insert_or_get_endpoint(identity) do
    case Repo.insert(
           CMS.ArtimentMentions.Model.Endpoint.changeset(
             %CMS.ArtimentMentions.Model.Endpoint{},
             identity
           )
         ) do
      {:ok, endpoint} ->
        {:ok, endpoint}

      {:error, changeset} ->
        case endpoint_query(identity) |> lock("FOR UPDATE") |> Repo.one() do
          %CMS.ArtimentMentions.Model.Endpoint{} = endpoint ->
            if endpoint_identity(endpoint) == {:ok, identity} do
              {:ok, endpoint}
            else
              {:error, :mention_endpoint_identity_conflict}
            end

          nil ->
            {:error, changeset}
        end
    end
  end

  @doc "Persists one complete immutable parser result for a source token and parser version."
  @spec save_version_facts(Ecto.UUID.t(), String.t(), pos_integer(), [map()]) ::
          {:ok, map()} | {:error, term()}
  def save_version_facts(source_endpoint_id, source_token, parser_version, occurrences)
      when is_binary(source_endpoint_id) and is_binary(source_token) and
             is_integer(parser_version) and parser_version > 0 and is_list(occurrences) do
    occurrences = normalize_occurrence_times(occurrences)

    with {:ok, :pass} <- validate_occurrence_times(occurrences) do
      Repo.transaction(fn ->
        endpoint =
          from(endpoint in CMS.ArtimentMentions.Model.Endpoint,
            where: endpoint.id == ^source_endpoint_id,
            lock: "FOR UPDATE"
          )
          |> Repo.one()

        if is_nil(endpoint), do: Repo.rollback(:source_endpoint_not_found)

        case Repo.get_by(CMS.ArtimentMentions.Model.MentionSet,
               source_endpoint_id: source_endpoint_id,
               source_token: source_token,
               parser_version: parser_version
             ) do
          %CMS.ArtimentMentions.Model.MentionSet{} = existing ->
            existing_occurrences =
              from(occurrence in CMS.ArtimentMentions.Model.Occurrence,
                where: occurrence.set_id == ^existing.id
              )
              |> Repo.all()

            if fact_signatures(existing_occurrences) == fact_signatures(occurrences) do
              existing
            else
              Repo.rollback(:immutable_mention_set_conflict)
            end

          nil ->
            set =
              case Repo.insert(
                     CMS.ArtimentMentions.Model.MentionSet.changeset(
                       %CMS.ArtimentMentions.Model.MentionSet{},
                       %{
                         source_endpoint_id: source_endpoint_id,
                         source_token: source_token,
                         parser_version: parser_version
                       }
                     )
                   ) do
                {:ok, set} -> set
                {:error, changeset} -> Repo.rollback(changeset)
              end

            Enum.each(occurrences, fn attrs ->
              attrs = Map.put(attrs, :set_id, set.id)

              case Repo.insert(
                     CMS.ArtimentMentions.Model.Occurrence.changeset(
                       %CMS.ArtimentMentions.Model.Occurrence{},
                       attrs
                     )
                   ) do
                {:ok, _occurrence} -> {:ok, :pass}
                {:error, changeset} -> Repo.rollback(changeset)
              end
            end)

            set
        end
      end)
      |> version_fact_result()
    end
  end

  def save_version_facts(_source_endpoint_id, _source_token, _parser_version, _occurrences) do
    {:error, :invalid_mention_set}
  end

  defp validate_occurrence_times(occurrences) do
    inconsistent? =
      occurrences
      |> Enum.group_by(&occurrence_edge_identity/1)
      |> Enum.any?(fn {_edge, facts} ->
        times = Enum.map(facts, &Map.get(&1, :mentioned_at)) |> Enum.uniq()
        length(times) != 1 or not match?(%DateTime{}, hd(times))
      end)

    if inconsistent? do
      {:error, :inconsistent_mentioned_at}
    else
      {:ok, :pass}
    end
  end

  defp normalize_occurrence_times(occurrences) do
    Enum.map(occurrences, fn
      %{mentioned_at: %DateTime{} = mentioned_at} = occurrence ->
        Map.put(occurrence, :mentioned_at, DateTime.truncate(mentioned_at, :second))

      occurrence ->
        occurrence
    end)
  end

  defp fact_signatures(facts) do
    facts
    |> Enum.map(fn fact ->
      signature = Map.new(@version_fact_fields, &{&1, Map.get(fact, &1)})
      Map.update!(signature, :mentioned_at, &DateTime.to_unix(&1, :second))
    end)
    |> Enum.sort_by(&Map.get(&1, :occurrence_key))
  end

  defp version_fact_result({:ok, set}), do: {:ok, set}
  defp version_fact_result({:error, reason}), do: {:error, reason}
end
