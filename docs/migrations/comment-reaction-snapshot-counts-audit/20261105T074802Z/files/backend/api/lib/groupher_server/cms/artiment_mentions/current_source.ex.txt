defmodule GroupherServer.CMS.ArtimentMentions.CurrentSource do
  @moduledoc """
  Reads the canonical current selection for a stable Mention source.

      endpoint -> current public/body token -> rebuild or eligible delivery
  """
  import Ecto.Query
  alias GroupherServer.{CMS, Repo}

  alias CMS.Model.{
    Article,
    ArticlePublic,
    Comment,
    CommentBody,
    DocPublic,
    DocBranch,
    DocBranchVersion
  }

  def load(%{type: :comment, comment_id: id}) do
    with %Comment{} = comment <- Repo.get(Comment, id),
         %CommentBody{} = body <- Repo.get_by(CommentBody, comment_id: id),
         %Article{} = parent <- Repo.get(Article, comment.article_id),
         {:ok, selected} <- load(%{type: parent.thread, article_id: parent.id}) do
      {:ok,
       %{
         content: comment,
         token: "#{id}:#{body.version}:#{body.body_hash}",
         title: selected.title,
         article_id: parent.id,
         thread: parent.thread,
         author_user_id: comment.author_id,
         branch_id: selected.branch_id
       }}
    else
      _ -> {:error, :mention_current_source_missing}
    end
  end

  def load(%{type: :doc, article_id: id}) do
    query =
      from(public in DocPublic,
        join: branch in DocBranch,
        on: branch.id == public.branch_id and branch.type == :main,
        join: version in DocBranchVersion,
        on: version.id == public.branch_version_id,
        where: public.article_id == ^id,
        select: {public, version.revision_id}
      )

    case Repo.all(query) do
      [{public, revision_id}] ->
        with %Article{thread: :doc} = article <- Repo.get(Article, id) do
          content = %{
            article_id: id,
            thread: :doc,
            branch_id: public.branch_id,
            revision_id: revision_id,
            title: public.title
          }

          {:ok,
           %{
             content: content,
             token:
               "#{public.branch_id}:#{public.branch_version_id}:#{public.publication_version}",
             title: public.title,
             article_id: id,
             thread: :doc,
             author_id: article.author_id,
             branch_id: public.branch_id
           }}
        else
          _ -> {:error, :mention_current_source_missing}
        end

      [] ->
        {:error, :mention_current_source_missing}

      _ ->
        {:error, :mention_current_source_ambiguous}
    end
  end

  def load(%{type: type, article_id: id}) when type in [:post, :blog, :changelog, :ticket] do
    with %Article{thread: ^type} = article <- Repo.get(Article, id),
         %ArticlePublic{} = public <- Repo.get(ArticlePublic, id) do
      {:ok,
       %{
         content: article,
         token: "#{public.revision_id}:#{public.publication_version}",
         title: public.title,
         article_id: id,
         thread: type,
         author_id: article.author_id,
         branch_id: nil
       }}
    else
      _ -> {:error, :mention_current_source_missing}
    end
  end

  def load(_), do: {:error, :invalid_mention_source}
end
