defmodule GroupherServer.CMS.ArtimentMentions.MaintenanceAuthorization do
  @moduledoc """
  Checks the Auth-issued capability required by Mentions initialization and saved-set rebuilds.

      Auth service JWT -> ServiceAuth.Verifier -> exact maintenance audience + scope
  """

  alias GroupherServerWeb.ServiceAuth.Verifier

  @audience "phoenix:cms-maintenance"
  @scopes %{initialize: "cms:mentions:initialize", rebuild: "cms:mentions:rebuild"}

  @spec verify(binary()) :: {:ok, map()} | {:error, :mentions_maintenance_unauthorized}
  def verify(token), do: verify(token, :initialize)

  def verify(token, operation) when is_binary(token) and operation in [:initialize, :rebuild] do
    required_scope = Map.fetch!(@scopes, operation)

    with {:ok, claims} <- Verifier.verify(token),
         @audience <- claims.audience,
         true <- MapSet.member?(claims.scopes, required_scope) do
      {:ok, Map.take(claims, [:subject, :token_id])}
    else
      _ -> {:error, :mentions_maintenance_unauthorized}
    end
  end

  def verify(_, _), do: {:error, :mentions_maintenance_unauthorized}
end
