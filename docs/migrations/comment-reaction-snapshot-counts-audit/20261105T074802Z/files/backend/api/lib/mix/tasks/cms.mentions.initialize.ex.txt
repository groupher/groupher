defmodule Mix.Tasks.Cms.Mentions.Initialize do
  @moduledoc """
  Local Mix entry for the authenticated Mentions initializer.

      Mix task arguments
        -> initializer CLI
        -> maintenance capability check
        -> bounded report / preflight / apply

  Production uses the no-Mix release CLI; both entries share one implementation.
  """

  use Mix.Task

  alias GroupherServer.CMS.ArtimentMentions.InitializeCLI

  @shortdoc "Initialize current Mentions sets under a maintenance capability"

  @impl Mix.Task
  def run(args) do
    Mix.Task.run("app.start")

    try do
      result = InitializeCLI.run(args)
      Mix.shell().info(Jason.encode!(result, pretty: true))
    rescue
      error -> Mix.raise(Exception.message(error))
    end
  end
end
