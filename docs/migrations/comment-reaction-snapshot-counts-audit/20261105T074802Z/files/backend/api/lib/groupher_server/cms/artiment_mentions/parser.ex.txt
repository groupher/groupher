defmodule GroupherServer.CMS.ArtimentMentions.Parser do
  @moduledoc """
  Extracts mention candidates from the canonical Plate AST.

  The parser walks blocks and inline children, preserves the block/path location
  for each occurrence, and normalizes both editor mention nodes and pasted links
  into the same downstream shape.

      Plate AST
          |
          v
      inline mention nodes + text/link URLs
          |
          v
      internal candidates  ----> resolved article/comment/user structs
      external URLs        ----> url + hash facts

  This module only parses and resolves mention targets. It does not delete,
  insert, notify, or update mailbox state; those side effects belong to the
  mention sync and messaging layers.
  """

  import Ecto.Query, warn: false
  import GroupherServer.CMS.Artiment.Matcher

  alias GroupherServer.{Accounts, CMS, Repo}

  alias Accounts.Model.User

  alias CMS.{
    ArtimentMentions.Config,
    ArtimentMentions.PublicUrl,
    ErrorCat,
    FrontDesk
  }

  alias CMS.Gate.Context.Scope.Article, as: ArticleScope
  alias CMS.Gate.Context.Scope.Comment, as: CommentScope
  alias CMS.Gate.Context.Scope.Doc, as: DocScope
  alias CMS.Gate.Context.Scope.Ticket, as: TicketScope

  alias CMS.Model.{
    ArticleBinding,
    Article,
    ArticleRevision,
    Comment,
    Community,
    DocBranch,
    DocBranchVersion,
    TicketState
  }

  @threads Config.threads()
  @max_url_bytes 1200
  @valid_article_prefix Config.valid_article_prefixes()
  @href_regex ~r/<a\b[^>]*\bhref\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+))/i
  @url_regex ~r/https?:\/\/[^\s<>"']+/u

  @type parsed_mention :: %{
          required(:mentioned_scope) => :internal | :external,
          required(:mentioned_type) => atom(),
          optional(:mentioned_id) => integer(),
          optional(:mentioned_url) => String.t(),
          optional(:mentioned_url_hash) => String.t(),
          required(:mention_case) => :inline_mention | :link,
          required(:occurrence) => map(),
          optional(:artiment) => struct(),
          optional(:parent_article) => struct()
        }

  @doc """
  Parses the canonical Plate AST and batch-resolves all internal Mention targets.

  Comment results retain their already-loaded parent Article so downstream
  snapshot construction stays query-free.
  """
  @spec parse(list(), keyword()) :: [parsed_mention()]
  def parse(ast, opts \\ [])

  def parse(ast, opts) when is_list(ast) do
    ast
    |> candidates()
    |> resolve_internal_mentions(Keyword.get(opts, :community_id))
    |> Enum.uniq()
  end

  def parse(_, _opts), do: []

  @doc "Extracts ordered Mention candidates before resolving current target visibility."
  @spec candidates(list()) :: [map()]
  def candidates(ast) when is_list(ast) do
    ast
    |> Enum.with_index()
    |> Enum.flat_map(fn {node, index} ->
      block_id = node_block_id(node)
      collect_mentions_from_node(node, block_id, [index])
    end)
  end

  def candidates(_), do: []

  @doc "Rejects stable IDs whose canonical resource type contradicts the mention key."
  def validate_identity_types(ast) when is_list(ast) do
    candidates = candidates(ast)

    ids =
      candidates
      |> Enum.filter(&(Map.get(&1, :type) in @threads))
      |> Enum.flat_map(fn candidate ->
        case Ecto.UUID.cast(Map.get(candidate, :value)) do
          {:ok, id} -> [id]
          _ -> []
        end
      end)
      |> Enum.uniq()

    types =
      case ids do
        [] ->
          %{}

        _ ->
          from(a in Article, where: a.id in ^ids, select: {a.id, a.thread})
          |> Repo.all()
          |> Map.new()
      end

    mismatched? =
      Enum.any?(candidates, fn candidate ->
        case Map.get(types, Map.get(candidate, :value)) do
          nil -> false
          type -> type != Map.get(candidate, :type)
        end
      end)

    if mismatched? do
      {:error, :mention_target_type_mismatch}
    else
      validate_locator_types(candidates)
    end
  end

  defp validate_locator_types(candidates) do
    locators =
      Enum.flat_map(candidates, fn candidate ->
        case Map.get(candidate, :value) do
          {:public, slug, inner_id} -> [{slug, inner_id, candidate.type}]
          {:public_comment, slug, thread, inner_id, _comment_id} -> [{slug, inner_id, thread}]
          _ -> []
        end
      end)
      |> Enum.uniq()

    if locators == [] do
      {:ok, :pass}
    else
      filter =
        Enum.reduce(locators, dynamic(false), fn {slug, inner_id, _type}, acc ->
          dynamic(
            [_article, binding, community],
            ^acc or
              (community.slug == ^slug and binding.inner_id == ^inner_id)
          )
        end)

      actual =
        from(article in Article,
          join: binding in ArticleBinding,
          on: binding.article_id == article.id,
          join: community in Community,
          on: community.id == binding.community_id,
          where: ^filter,
          select: {{community.slug, binding.inner_id}, article.thread}
        )
        |> Repo.all()
        |> Map.new()

      mismatched? =
        Enum.any?(locators, fn {slug, inner_id, type} ->
          case Map.get(actual, {slug, inner_id}) do
            nil -> false
            ^type -> false
            _ -> true
          end
        end)

      if mismatched? do
        {:error, :mention_target_type_mismatch}
      else
        {:ok, :pass}
      end
    end
  end

  @doc "Rejects URLs that cannot fit safely in the indexed current projection."
  @spec validate_url_lengths(list()) :: {:ok, :pass} | {:error, term()}
  def validate_url_lengths(ast) when is_list(ast) do
    too_long? =
      ast
      |> candidates()
      |> Enum.any?(fn candidate ->
        case Map.get(candidate, :mentioned_url) do
          url when is_binary(url) -> byte_size(url) > @max_url_bytes
          _ -> false
        end
      end)

    if too_long?, do: {:error, ErrorCat.mention_url_too_long()}, else: {:ok, :pass}
  end

  def validate_url_lengths(_), do: {:error, :invalid_mention_ast}

  @doc "Resolves candidates against stable identities for the authorized initializer only."
  @spec parse_for_initialization(list()) ::
          {:ok, [parsed_mention()]} | {:error, term()}
  def parse_for_initialization(ast) when is_list(ast) do
    candidates = candidates(ast)

    resolved =
      candidates
      |> Enum.reject(&ticket_public_url_candidate?/1)
      |> resolve_internal_mentions(nil)

    with {:ok, :pass} <- validate_identity_types(ast),
         {:ok, overrides, additions} <- resolve_stable_initialization_candidates(candidates) do
      resolved_keys =
        MapSet.new(resolved, &candidate_occurrence_key(Map.get(&1, :occurrence, %{})))

      mentions =
        Enum.map(resolved, fn mention ->
          key = candidate_occurrence_key(Map.get(mention, :occurrence, %{}))
          Map.get(overrides, key, mention)
        end)

      additions =
        Enum.reject(
          additions,
          &MapSet.member?(resolved_keys, candidate_occurrence_key(&1.occurrence))
        )

      {:ok, Enum.uniq(mentions ++ additions)}
    end
  end

  def parse_for_initialization(_), do: {:error, :invalid_mention_initialization_scope}

  defp resolve_stable_initialization_candidates(candidates) do
    stable_candidates = Enum.filter(candidates, &stable_initialization_candidate?/1)

    with {:ok, facts} <- load_stable_initialization_facts(stable_candidates) do
      stable_candidates
      |> Enum.reduce_while({:ok, %{}, []}, fn candidate, {:ok, overrides, additions} ->
        case resolve_initialization_candidate(candidate, facts) do
          {:ok, nil} ->
            {:cont, {:ok, overrides, additions}}

          {:ok, mentioned} ->
            resolved = resolved_internal_mention(candidate, mentioned)
            key = candidate_occurrence_key(candidate.occurrence)
            {:cont, {:ok, Map.put(overrides, key, resolved), [resolved | additions]}}

          {:error, reason} ->
            {:halt, {:error, reason}}
        end
      end)
      |> case do
        {:ok, overrides, additions} -> {:ok, overrides, Enum.reverse(additions)}
        {:error, _reason} = error -> error
      end
    end
  end

  defp stable_initialization_candidate?(%{
         internal_mention_candidate?: true,
         type: type,
         occurrence: occurrence
       }) do
    type in (@threads ++ [:comment]) and
      Map.get(occurrence, :normalized_from) in ["inline_mention", "link"]
  end

  defp stable_initialization_candidate?(_), do: false

  defp ticket_public_url_candidate?(%{
         type: :ticket,
         value: {:public, _community_slug, _inner_id},
         occurrence: %{normalized_from: "link"}
       }) do
    true
  end

  defp ticket_public_url_candidate?(_), do: false

  defp load_stable_initialization_facts(candidates) do
    public_article_locators =
      Enum.flat_map(candidates, fn
        %{type: type, value: {:public, _, _} = locator} when type in @threads ->
          [locator]

        %{type: :comment, value: {:public_comment, slug, _thread, inner_id, _comment_inner_id}} ->
          [{:public, slug, inner_id}]

        _ ->
          []
      end)
      |> Enum.uniq()

    slugs = Enum.map(public_article_locators, fn {:public, slug, _inner_id} -> slug end)

    communities =
      case slugs do
        [] ->
          %{}

        _ ->
          Repo.all(from(community in Community, where: community.slug in ^slugs))
          |> Map.new(&{&1.slug, &1})
      end

    binding_pairs =
      public_article_locators
      |> Enum.map(fn {:public, slug, inner_id} ->
        with %Community{id: community_id} <- Map.get(communities, slug),
             inner_id when is_integer(inner_id) <- cast_id(inner_id) do
          {community_id, inner_id}
        end
      end)
      |> Enum.reject(&is_nil/1)
      |> Enum.uniq()

    bindings =
      case binding_pairs do
        [] ->
          %{}

        _ ->
          binding_pairs
          |> binding_pair_filter()
          |> then(fn filter ->
            from(binding in ArticleBinding, where: ^filter)
          end)
          |> Repo.all()
          |> Enum.group_by(&{&1.community_id, &1.inner_id})
      end

    direct_article_ids =
      candidates
      |> Enum.filter(&(&1.type in @threads))
      |> Enum.map(&cast_article_id(&1.value))
      |> Enum.filter(&is_binary/1)

    direct_comment_ids =
      candidates
      |> Enum.filter(&(&1.type == :comment))
      |> Enum.reject(&match?({:public_comment, _, _, _, _}, &1.value))
      |> Enum.map(&cast_comment_identity(&1.value))
      |> Enum.flat_map(fn
        {:ok, id} -> [id]
        {:error, _reason} -> []
      end)
      |> Enum.uniq()

    comments =
      case direct_comment_ids do
        [] ->
          %{}

        _ ->
          Repo.all(from(comment in Comment, where: comment.id in ^direct_comment_ids))
          |> Map.new(&{&1.id, &1})
      end

    article_ids =
      (direct_article_ids ++
         Enum.flat_map(Map.values(bindings), &Enum.map(&1, fn binding -> binding.article_id end)) ++
         Enum.map(Map.values(comments), & &1.article_id))
      |> Enum.uniq()

    articles =
      case article_ids do
        [] ->
          %{}

        _ ->
          Repo.all(from(article in Article, where: article.id in ^article_ids))
          |> Map.new(&{&1.id, &1})
      end

    user_ticket_ids =
      case article_ids do
        [] ->
          MapSet.new()

        ids ->
          Repo.all(
            from(ticket in TicketState,
              where: ticket.article_id in ^ids and ticket.type == :user,
              select: ticket.article_id
            )
          )
          |> MapSet.new()
      end

    doc_branches =
      case article_ids do
        [] ->
          %{}

        ids ->
          from(version in DocBranchVersion,
            join: branch in DocBranch,
            on: branch.id == version.branch_id and branch.type == :main,
            where: version.article_id in ^ids,
            select: {version.article_id, branch.id, branch.community_id},
            distinct: true
          )
          |> Repo.all()
          |> Enum.group_by(&elem(&1, 0), fn {_article_id, branch_id, community_id} ->
            {branch_id, community_id}
          end)
      end

    public_comment_keys =
      public_comment_candidates(candidates)
      |> Enum.flat_map(fn {:public_comment, slug, thread, inner_id, comment_inner_id} ->
        with %Community{} = community <- Map.get(communities, slug),
             inner_id when is_integer(inner_id) <- cast_id(inner_id),
             comment_inner_id when is_integer(comment_inner_id) <- cast_id(comment_inner_id) do
          bindings
          |> Map.get({community.id, inner_id}, [])
          |> Enum.flat_map(fn binding ->
            case Map.get(articles, binding.article_id) do
              %Article{thread: ^thread} = article ->
                if thread != :doc or
                     Enum.any?(Map.get(doc_branches, article.id, []), fn {_branch, owner_id} ->
                       owner_id == community.id
                     end) do
                  [{article.id, thread, comment_inner_id}]
                else
                  []
                end

              _ ->
                []
            end
          end)
        else
          _ -> []
        end
      end)
      |> Enum.uniq()

    public_comments =
      case public_comment_keys do
        [] ->
          %{}

        _ ->
          public_comment_keys
          |> public_comment_filter()
          |> then(fn filter ->
            from(comment in Comment, where: ^filter)
          end)
          |> Repo.all()
          |> Enum.group_by(&{&1.article_id, &1.thread, &1.inner_id})
      end

    {:ok,
     %{
       communities: communities,
       bindings: bindings,
       articles: articles,
       comments: comments,
       user_ticket_ids: user_ticket_ids,
       doc_branches: doc_branches,
       public_comments: public_comments
     }}
  end

  defp public_comment_candidates(candidates) do
    candidates
    |> Enum.filter(fn
      %{type: :comment, value: {:public_comment, _, _, _, _}} -> true
      _ -> false
    end)
    |> Enum.map(& &1.value)
    |> Enum.uniq()
  end

  defp binding_pair_filter(pairs) do
    Enum.reduce(pairs, dynamic(false), fn {community_id, inner_id}, filter ->
      dynamic(
        [binding],
        ^filter or
          (binding.community_id == ^community_id and binding.inner_id == ^inner_id)
      )
    end)
  end

  defp public_comment_filter(keys) do
    Enum.reduce(keys, dynamic(false), fn {article_id, thread, inner_id}, filter ->
      dynamic(
        [comment],
        ^filter or
          (comment.article_id == ^article_id and comment.thread == ^thread and
             comment.inner_id == ^inner_id)
      )
    end)
  end

  defp valid_initialization_thread(%Article{thread: :ticket, id: id}, :ticket, facts) do
    if MapSet.member?(facts.user_ticket_ids, id) do
      {:ok, :pass}
    else
      {:error, :invalid_mention_ticket_identity}
    end
  end

  defp valid_initialization_thread(%Article{thread: thread}, thread, _facts) do
    {:ok, :pass}
  end

  defp valid_initialization_thread(_, _, _), do: {:error, :mention_target_type_mismatch}

  defp resolve_initialization_candidate(%{type: type, value: value}, facts)
       when type in @threads do
    case value do
      {:public, slug, inner_id} ->
        resolve_public_article_identity(type, slug, inner_id, facts)

      _ ->
        resolve_stable_article_identity(type, value, facts)
    end
  end

  defp resolve_initialization_candidate(%{type: :comment, value: value}, facts) do
    case value do
      {:public_comment, slug, thread, inner_id, comment_inner_id} ->
        with {:ok, %{article_id: _article_id} = parent} <-
               resolve_public_article_identity(thread, slug, inner_id, facts),
             %Comment{} = comment <-
               public_comment_from_facts(parent, thread, comment_inner_id, facts) do
          {:ok,
           %{
             type: :comment,
             id: comment.id,
             url: nil,
             artiment: comment,
             parent_article: %{
               article_id: parent.article_id,
               thread: thread
             }
           }}
        else
          nil -> {:ok, nil}
          {:ok, nil} -> {:ok, nil}
          {:error, _reason} = error -> error
        end

      _ ->
        with {:ok, comment_id} <- cast_comment_identity(value),
             %Comment{} = comment <- Map.get(facts.comments, comment_id),
             %Article{} = article <- Map.get(facts.articles, comment.article_id),
             true <- article.thread == comment.thread,
             {:ok, parent} <- initialize_comment_parent(article, comment, facts) do
          {:ok,
           %{
             type: :comment,
             id: comment.id,
             url: nil,
             artiment: comment,
             parent_article: parent
           }}
        else
          nil -> {:ok, nil}
          false -> {:error, :mention_comment_parent_mismatch}
          {:error, _reason} = error -> error
        end
    end
  end

  defp resolve_initialization_candidate(_, _), do: {:ok, nil}

  defp public_comment_from_facts(parent, thread, comment_inner_id, facts) do
    key = {
      parent.article_id,
      thread,
      cast_id(comment_inner_id)
    }

    case Map.get(facts.public_comments, key, []) do
      [] -> nil
      [comment] -> comment
      _ -> {:error, :ambiguous_mention_target_identity}
    end
  end

  defp resolve_public_article_identity(type, slug, inner_id, facts) do
    with %Community{} = community <- Map.get(facts.communities, slug),
         inner_id when is_integer(inner_id) <- cast_id(inner_id),
         bindings when is_list(bindings) <-
           Map.get(facts.bindings, {community.id, inner_id}, []),
         {:ok, %{article: article} = identity} <-
           public_article_binding_identity(type, community, bindings, facts) do
      {:ok,
       %{
         type: type,
         id: article.id,
         article_id: article.id,
         branch_id: identity.branch_id,
         community_id: community.id,
         url: nil,
         artiment: %{
           id: article.id,
           article_id: article.id,
           thread: article.thread,
           branch_id: identity.branch_id,
           community_id: community.id
         }
       }}
    else
      nil -> {:ok, nil}
      {:ok, nil} -> {:ok, nil}
      :error -> {:ok, nil}
      {:error, _reason} = error -> error
    end
  end

  defp public_article_binding_identity(type, community, bindings, facts) do
    identities =
      bindings
      |> Enum.flat_map(fn binding ->
        case Map.get(facts.articles, binding.article_id) do
          %Article{thread: ^type} = article ->
            branch_ids =
              if type == :doc do
                facts.doc_branches
                |> Map.get(article.id, [])
                |> Enum.filter(fn {_branch_id, community_id} ->
                  community_id == community.id
                end)
                |> Enum.map(&elem(&1, 0))
              else
                [nil]
              end

            Enum.map(branch_ids, &%{article: article, branch_id: &1})

          _ ->
            []
        end
      end)
      |> Enum.uniq_by(&{&1.article.id, &1.branch_id})

    case identities do
      [] ->
        {:ok, nil}

      [identity] ->
        case valid_initialization_thread(identity.article, type, facts) do
          {:ok, :pass} -> {:ok, identity}
          {:error, _reason} = error -> error
        end

      _ ->
        {:error, :ambiguous_mention_target_identity}
    end
  end

  defp resolve_stable_article_identity(type, value, facts) do
    with {:ok, article_id} <- Ecto.UUID.cast(value),
         %Article{thread: ^type} = article <- Map.get(facts.articles, article_id),
         {:ok, :pass} <- valid_initialization_thread(article, type, facts),
         {:ok, {branch_id, target_community_id}} <- initialization_branch(type, article.id, facts) do
      {:ok,
       %{
         type: type,
         id: article.id,
         article_id: article.id,
         branch_id: branch_id,
         community_id: target_community_id,
         url: nil,
         artiment: %{
           id: article.id,
           article_id: article.id,
           thread: article.thread,
           branch_id: branch_id,
           community_id: target_community_id
         }
       }}
    else
      nil -> {:ok, nil}
      :error -> {:ok, nil}
      {:error, _reason} = error -> error
    end
  end

  defp initialization_branch(:doc, article_id, facts) do
    case Map.get(facts.doc_branches, article_id, []) do
      [{branch_id, target_community_id}] -> {:ok, {branch_id, target_community_id}}
      [] -> :error
      _ -> {:error, :ambiguous_mention_target_identity}
    end
  end

  defp initialization_branch(_type, _article_id, _facts), do: {:ok, {nil, nil}}

  defp initialize_comment_parent(%Article{thread: :doc} = article, _comment, facts) do
    with {:ok, {_branch_id, owner_id}} <- initialization_branch(:doc, article.id, facts) do
      {:ok, %{article_id: article.id, thread: :doc, community_id: owner_id}}
    else
      :error -> {:error, :mention_comment_parent_mismatch}
      {:error, _reason} = error -> error
    end
  end

  defp initialize_comment_parent(%Article{} = article, _comment, _facts) do
    {:ok, %{article_id: article.id, thread: article.thread}}
  end

  defp cast_comment_identity(value) when is_integer(value) and value > 0, do: {:ok, value}

  defp cast_comment_identity(value) when is_binary(value) do
    case Integer.parse(value) do
      {id, ""} when id > 0 -> {:ok, id}
      _ -> {:error, :invalid_mention_comment_identity}
    end
  end

  defp cast_comment_identity(_), do: {:error, :invalid_mention_comment_identity}

  defp candidate_occurrence_key(occurrence) do
    "#{Map.get(occurrence, :block_id) || "block"}:#{Enum.join(Map.get(occurrence, :path, []), ".")}"
  end

  defp collect_mentions_from_node(%{"type" => "mention"} = node, block_id, path) do
    node
    |> parse_inline_mention(block_id, path)
    |> List.wrap()
    |> Enum.reject(&is_nil/1)
    |> Kernel.++(collect_children_mentions(node, block_id, path))
  end

  defp collect_mentions_from_node(%{"type" => type} = node, block_id, path)
       when type in ["a", "link"] do
    href = Map.get(node, "url") || Map.get(node, "href")

    link_mentions =
      if is_binary(href) and href != "" do
        parse_link_mention(href, block_id, path)
      else
        []
      end

    child_mentions =
      collect_children_mentions(node, block_id, path)
      |> Enum.reject(fn candidate ->
        link_mentions != [] and get_in(candidate, [:occurrence, :raw_url]) == href
      end)

    link_mentions ++ child_mentions
  end

  defp collect_mentions_from_node(%{"text" => text} = node, block_id, path)
       when is_binary(text) do
    text_mentions =
      text
      |> extract_links_from_text()
      |> Enum.flat_map(&parse_link_mention(&1, block_id, path))

    text_mentions ++ collect_children_mentions(node, block_id, path)
  end

  defp collect_mentions_from_node(node, block_id, path) when is_map(node) do
    collect_children_mentions(node, block_id, path)
  end

  defp collect_mentions_from_node(_, _, _), do: []

  defp collect_children_mentions(%{"children" => children}, block_id, path)
       when is_list(children) do
    children
    |> Enum.with_index()
    |> Enum.flat_map(fn {node, index} ->
      collect_mentions_from_node(node, block_id, path ++ [index])
    end)
  end

  defp collect_children_mentions(_, _, _), do: []

  defp parse_inline_mention(node, block_id, path) do
    with {:ok, type} <- inline_mention_type(node),
         value when not is_nil(value) <- inline_mention_value(type, node) do
      internal_mention_candidate(type, value, %{
        mention_case: :inline_mention,
        editor_target_id: Map.get(node, "key"),
        block_id: block_id,
        path: path,
        display: inline_mention_display(node),
        normalized_from: "inline_mention"
      })
    else
      _ -> nil
    end
  end

  defp inline_mention_type(%{"key" => key}) when is_binary(key) do
    case String.split(key, ":", parts: 2) do
      [type, _id] -> parse_mention_type(type)
      _ -> {:error, ErrorCat.invalid_mention_type()}
    end
  end

  defp inline_mention_type(_), do: {:error, ErrorCat.invalid_mention_type()}

  defp inline_mention_value(:user, %{"key" => "user:" <> login}), do: login
  defp inline_mention_value(:user, _), do: nil

  defp inline_mention_value(_type, %{"key" => key}) when is_binary(key) do
    case String.split(key, ":", parts: 2) do
      [_type, id] when id != "" -> id
      _ -> nil
    end
  end

  defp inline_mention_value(_, _), do: nil

  defp inline_mention_display(%{"children" => children} = node) when is_list(children) do
    display =
      children
      |> Enum.map(&extract_text/1)
      |> Enum.reject(&(&1 == ""))
      |> Enum.join(" ")

    if display == "", do: Map.get(node, "value", ""), else: display
  end

  defp inline_mention_display(%{"value" => value}) when is_binary(value), do: value
  defp inline_mention_display(_), do: ""

  defp extract_text(%{"text" => text}) when is_binary(text), do: text

  defp extract_text(%{"children" => children}) when is_list(children) do
    Enum.map_join(children, " ", &extract_text/1)
  end

  defp extract_text(_), do: ""

  defp parse_link_mention(url, block_id, path) do

    |> classify_url()
    |> do_parse_link_mention(url, block_id, path)

    |> Enum.reject(&is_nil/1)
  end

  defp classify_url(url) do
    case public_article_link(url) do
      {:ok, mention} ->
        {:ok, mention}

      :external ->
        if site_article_link?(url) do
          internal_mention_from_url(url)
        else
          {:ok, %{scope: :external, url: url, hash: hash_url(url)}}
        end
    end
  end

  # The host is part of the locator's trust boundary. A custom domain has no
  # backend host registry, so it remains an external URL even if its path fits.
  defp public_article_link(url) do
    uri = URI.parse(url)
    site = URI.parse(Config.site_host())

    trusted_host? =
      (uri.scheme == site.scheme and uri.host == site.host and uri.port == site.port) or
        (is_nil(uri.scheme) and is_nil(uri.host) and is_binary(uri.path) and
           String.starts_with?(uri.path, "/"))

    if trusted_host? and is_nil(uri.userinfo) do
      case String.split(uri.path || "", "/", trim: true) do
        [community, thread_slug, inner_id | rest] when rest == [] or length(rest) == 1 ->
          with {:ok, thread} <- parse_thread_slug(thread_slug),
               {inner, ""} when inner > 0 <- Integer.parse(inner_id),
               true <- thread == :doc or rest == [] do
            case uri.fragment do
              nil ->
                {:ok,
                 %{
                   scope: :internal,
                   type: thread,
                   value: {:public, community, inner}
                 }}

              "comment-" <> comment_inner_id ->
                case Integer.parse(comment_inner_id) do
                  {comment_inner, ""} when comment_inner > 0 ->
                    {:ok,
                     %{
                       scope: :internal,
                       type: :comment,
                       value: {:public_comment, community, thread, inner, comment_inner}
                     }}

                  _ ->
                    :external
                end

              _ ->
                :external
            end
          else
            _ -> :external
          end

        _ ->
          :external
      end
    else
      :external
    end
  end

  defp internal_mention_from_url(url) do
    case link_for_comment?(url) do
      true -> comment_mention_from_url(url)
      false -> article_mention_from_url(url)
    end
  end

  defp comment_mention_from_url(url) do
    %{query: query} = URI.parse(url)

    try do
      comment_id = URI.decode_query(query || "") |> Map.get("comment_id")

      case comment_id do
        nil -> {:error, ErrorCat.invalid_comment_link()}
        comment_id -> {:ok, %{scope: :internal, type: :comment, value: comment_id}}
      end
    rescue
      _ -> {:error, ErrorCat.invalid_comment_link()}
    end
  end

  defp article_mention_from_url(url) do
    %{path: path} = URI.parse(url)
    path_list = String.split(path || "", "/")
    article_id = Enum.at(path_list, 2)

    with {:ok, thread} <- parse_thread_slug(Enum.at(path_list, 1)),
         article_id when not is_nil(article_id) <- article_id do
      {:ok, %{scope: :internal, type: thread, value: article_id}}
    end
  end

  defp internal_mention_candidate(type, value, occurrence) do
    %{
      internal_mention_candidate?: true,
      type: type,
      value: value,
      occurrence: occurrence
    }
  end

  defp resolve_internal_mentions(mentions, community_id) do
    cache = internal_mention_cache(mentions, community_id)

    mentions
    |> Enum.flat_map(fn
      %{internal_mention_candidate?: true} = candidate ->
        candidate
        |> resolve_internal_mention(cache)
        |> List.wrap()
        |> Enum.reject(&is_nil/1)

      mention ->
        [mention]
    end)
  end

  defp internal_mention_cache(mentions, _source_community_id) do
    candidates = Enum.filter(mentions, &Map.get(&1, :internal_mention_candidate?))
    comments = load_comments(candidates)

    %{
      users_by_login: load_users_by_login(candidates),
      articles_by_thread: load_articles_by_thread(candidates),
      comments: comments
    }
  end

  defp resolve_internal_mention(
         %{type: :user, value: login} = candidate,
         %{users_by_login: users_by_login}
       )
       when is_binary(login) do
    resolve_user(candidate, Map.get(users_by_login, login))
  end

  defp resolve_internal_mention(%{type: :user}, _), do: nil

  defp resolve_internal_mention(%{type: :comment, value: value} = candidate, cache) do
    with id when not is_nil(id) <- comment_locator(value),
         cache_key <- comment_cache_key(value),
         %{comment: comment, article: article} <- Map.get(cache.comments, cache_key),
         url when is_binary(url) <- PublicUrl.comment(article, comment.inner_id) do
      candidate
      |> resolved_internal_mention(%{
        type: :comment,
        id: comment.id,
        url: url,
        snapshot: %{title: article.title},
        artiment: comment,
        parent_article: article
      })
    else
      _ -> nil
    end
  end

  defp resolve_internal_mention(%{type: type, value: value} = candidate, cache)
       when type in @threads do
    with articles <- Map.get(cache.articles_by_thread, type, %{}),
         key <- article_cache_key(value),
         article when not is_nil(article) <- Map.get(articles, key),
         url when is_binary(url) <- PublicUrl.article(article) do
      candidate
      |> resolved_internal_mention(%{
        type: type,
        id: article.id,
        url: url,
        snapshot: %{title: article.title},
        artiment: article
      })
    else
      _ -> nil
    end
  end

  defp resolve_internal_mention(_, _), do: nil

  defp resolved_internal_mention(candidate, mentioned) do
    occurrence =
      case Map.get(candidate.occurrence, :normalized_from) do
        "link" ->
          Map.put(
            candidate.occurrence,
            :display,
            get_in(mentioned, [:snapshot, :title]) || Map.get(candidate.occurrence, :raw_url, "")
          )

        _ ->
          candidate.occurrence
      end

    mention_case =
      if Map.get(candidate.occurrence, :normalized_from) == "link",
        do: :link,
        else: :inline_mention

    %{
      mentioned_scope: :internal,
      mentioned_type: mentioned.type,
      mentioned_id: mentioned.id,
      mentioned_url: mentioned.url,
      mention_case: mention_case,
      artiment: mentioned.artiment,
      occurrence: occurrence
    }
    |> maybe_put_parent_article(mentioned)
  end

  defp maybe_put_parent_article(mention, %{parent_article: article}) do
    Map.put(mention, :parent_article, article)
  end

  defp maybe_put_parent_article(mention, _), do: mention

  defp resolve_user(candidate, %User{} = user) do
    candidate
    |> resolved_internal_mention(%{
      type: :user,
      id: user.id,
      url: Config.user_url(user.login),
      artiment: user
    })
  end

  defp load_users_by_login(candidates) do
    logins =
      candidates
      |> Enum.filter(&(&1.type == :user and is_binary(&1.value)))
      |> Enum.map(& &1.value)
      |> Enum.uniq()

    case logins do
      [] ->
        %{}

      logins ->
        User
        |> where([u], u.login in ^logins)
        |> Repo.all()
        |> Map.new(&{&1.login, &1})
    end
  end

  defp load_articles_by_thread(candidates) do
    @threads
    |> Map.new(fn thread ->
      thread_candidates = Enum.filter(candidates, &(&1.type == thread))

      stable_ids =
        thread_candidates
        |> Enum.filter(&is_binary(&1.value))
        |> Enum.map(&cast_article_id(&1.value))
        |> Enum.reject(&is_nil/1)
        |> Enum.uniq()

      public_locators =
        candidates
        |> Enum.filter(&(&1.type == thread))
        |> Enum.map(& &1.value)
        |> Enum.filter(&match?({:public, _, _}, &1))
        |> Enum.uniq()

      {thread,
       Map.merge(
         load_public_articles(thread, public_locators),
         load_stable_articles(thread, stable_ids)
       )}
    end)
  end

  defp load_stable_articles(_thread, []), do: %{}

  defp load_stable_articles(:ticket, ids) do
    pairs =
      Repo.all(
        from(binding in ArticleBinding,
          where: binding.article_id in ^ids,
          select: {binding.article_id, binding.community_id, nil}
        )
      )

    :ticket
    |> load_articles_by_identity_pairs(pairs)
    |> Map.values()
    |> Enum.sort_by(&{&1.id, &1.community_id})
    |> Enum.reduce(%{}, fn article, acc -> Map.put_new(acc, {article.id, :stable}, article) end)
  end

  defp load_stable_articles(thread, ids) do
    with %Ecto.Query{} = query <- public_article_scope(thread) do
      query
      |> where([article], article.id in ^ids)
      |> public_article_projection(thread)
      |> Repo.all()
      |> Enum.sort_by(&{&1.id, &1.community_id})
      |> Enum.reduce(%{}, fn article, acc -> Map.put_new(acc, {article.id, :stable}, article) end)
    else
      _ -> %{}
    end
  end

  defp load_public_articles(_thread, []), do: %{}

  defp load_public_articles(:ticket, locators) do
    Enum.reduce(locators, %{}, fn {:public, slug, inner_id} = locator, acc ->
      with %Community{} = community <- Repo.get_by(Community, slug: slug),
           {:ok, %{article: article, title: title, binding: binding}} <-
             CMS.Tickets.get_public(community, inner_id) do
        Map.put(acc, locator, %{
          id: article.id,
          article_id: article.id,
          branch_id: nil,
          thread: :ticket,
          community_id: community.id,
          community_slug: community.slug,
          inner_id: binding.inner_id,
          title: title
        })
      else
        _ -> acc
      end
    end)
  end

  defp load_public_articles(thread, locators) do
    with %Ecto.Query{} = query <- public_article_scope(thread),
         pair_filter <- public_locator_filter(locators),
         query <- where(query, ^pair_filter),
         query <- public_article_projection(query, thread),
         rows <- Repo.all(query) do
      Map.new(rows, &{{:public, &1.community_slug, &1.inner_id}, &1})
    else
      _ -> %{}
    end
  end

  defp load_articles_by_identity_pairs(_thread, []), do: %{}

  defp load_articles_by_identity_pairs(:ticket, pairs) do
    community_ids = pairs |> Enum.map(&elem(&1, 1)) |> Enum.uniq()

    communities =
      Repo.all(from(community in Community, where: community.id in ^community_ids))

    Enum.reduce(communities, %{}, fn community, resolved ->
      article_ids =
        pairs
        |> Enum.filter(&(elem(&1, 1) == community.id))
        |> Enum.map(&elem(&1, 0))
        |> Enum.uniq()

      with {:ok, %Ecto.Query{} = query} <-
             CMS.Gate.scope(TicketState, nil, :list, TicketScope.public(community)) do
        rows =
          from(
            [ticket_state: state, ticket_binding: binding, ticket_article: article] in query,
            join: revision in ArticleRevision,
            on: revision.id == as(:ticket_public).revision_id,
            where: state.article_id in ^article_ids,
            select: %{
              id: article.id,
              article_id: article.id,
              branch_id: nil,
              thread: article.thread,
              inner_id: binding.inner_id,
              slug: nil,
              title: revision.title
            }
          )
          |> Repo.all()

        Map.merge(
          resolved,
          Map.new(rows, fn row ->
            row = Map.merge(row, %{community_id: community.id, community_slug: community.slug})
            {{row.article_id, community.id, nil}, row}
          end)
        )
      else
        _ -> resolved
      end
    end)
  end

  defp load_articles_by_identity_pairs(:doc, pairs) do
    query = public_article_scope(:doc)

    filter =
      Enum.reduce(pairs, dynamic(false), fn {article_id, community_id, _branch_id}, acc ->
        dynamic(
          [article, gate_article_binding: binding],
          ^acc or
            (article.id == ^article_id and binding.community_id == ^community_id)
        )
      end)

    query
    |> where(^filter)
    |> public_article_projection(:doc)
    |> Repo.all()
    |> Map.new(fn article ->
      {{article.id, article.community_id, nil}, article}
    end)
  end

    |> where(^filter)

  defp public_article_projection(query, :doc) do
    from(
      [
        article,
        gate_article_binding: binding,
        gate_community: community,
        gate_article_public: public,
        gate_doc_branch: branch
      ] in query,
      join: version in DocBranchVersion,
      on: version.id == public.branch_version_id,
      join: revision in ArticleRevision,
      on: revision.id == version.revision_id,
      select: %{
        id: article.id,
        article_id: article.id,
        branch_id: branch.id,
        thread: article.thread,
        community_id: community.id,
        community_slug: community.slug,
        inner_id: binding.inner_id,
        slug: public.slug,
        title: revision.title
      }
    )
  end

  defp public_article_projection(query, _thread) do
    from(
      [
        article,
        gate_article_binding: binding,
        gate_community: community,
        gate_article_public: public
      ] in query,
      join: revision in ArticleRevision,
      on: revision.id == public.revision_id,
      select: %{
        id: article.id,
        article_id: article.id,
        branch_id: nil,
        thread: article.thread,
        community_id: community.id,
        community_slug: community.slug,
        inner_id: binding.inner_id,
        slug: public.slug,
        title: revision.title
      }
    )
  end

  defp public_locator_filter(locators) do
    Enum.reduce(locators, dynamic(false), fn {:public, slug, inner_id}, filter ->
      dynamic(
        [gate_article_binding: binding, gate_community: community],
        ^filter or (community.slug == ^slug and binding.inner_id == ^inner_id)
      )
    end)
  end

  defp load_comments(candidates) do
    ids =
      candidates
      |> Enum.filter(&(&1.type == :comment))
      |> Enum.map(&cast_id(&1.value))
      |> Enum.reject(&is_nil/1)
      |> Enum.uniq()

    public_pairs = load_public_comment_pairs(ids)
    comments = Enum.map(public_pairs, &elem(&1, 0))

    comments_by_id = Map.new(comments, &{&1.id, &1})

    comment_pairs =
      public_pairs
      |> Enum.map(fn {comment, community_id} ->
        {comment.id, community_id}
      end)
      |> Enum.uniq()

    articles_by_thread =
      comments
      |> comment_article_ids_by_thread()
      |> Map.new(fn {thread, _article_ids} ->
        pairs =
          comment_pairs
          |> Enum.flat_map(fn {comment_id, community_id} ->
            case comments_by_id[comment_id] do
              %Comment{thread: ^thread, article_id: article_id} ->
                [{article_id, community_id, nil}]

              _ ->
                []
            end
          end)
          |> Enum.uniq()

        {thread, load_articles_by_identity_pairs(thread, pairs)}
      end)

    id_comments =
      Enum.reduce(comment_pairs, %{}, fn {comment_id, community_id}, acc ->
        with %Comment{} = comment <- comments_by_id[comment_id],
             {:ok, thread} <- FrontDesk.thread_of(comment),
             {:ok, info} <- match(thread),
             article_id when not is_nil(article_id) <- Map.get(comment, info.foreign_key),
             articles <- Map.get(articles_by_thread, thread, %{}),
             article when not is_nil(article) <-
               Map.get(
                 articles,
                 {article_id, community_id, nil}
               ) do
          value = %{
            comment: comment,
            article: article,
            thread: thread
          }

          Map.put_new(acc, {comment.id, :stable}, value)
        else
          _ -> acc
        end
      end)

    Map.merge(id_comments, load_public_comments(candidates))
  end

  defp article_cache_key({:public, _, _} = locator), do: locator
  defp article_cache_key(value), do: {cast_article_id(value), :stable}

  defp comment_cache_key({:public_comment, _, _, _, _} = locator), do: locator
  defp comment_cache_key(value), do: {cast_id(value), :stable}

  defp load_public_comment_pairs([]), do: []

  defp load_public_comment_pairs(ids) do
    Comment
    |> CMS.Gate.scope(nil, :list, CommentScope.all_public())
    |> case do
      %Ecto.Query{} = query ->
        query
        |> where([comment], comment.id in ^ids and as(:gate_comment_lifecycle).state == :visible)
        |> order_by([comment, gate_comment_binding: binding], [comment.id, binding.community_id])
        |> select([comment, gate_comment_binding: binding], {comment, binding.community_id})
        |> Repo.all()

      {:error, _reason} ->
        []
    end
  end

  defp load_public_comments(candidates) do
    locators =
      candidates
      |> Enum.filter(&(&1.type == :comment))
      |> Enum.map(& &1.value)
      |> Enum.filter(&match?({:public_comment, _, _, _, _}, &1))
      |> Enum.uniq()

    case locators do
      [] ->
        %{}

      _ ->
        filter =
          Enum.reduce(locators, dynamic(false), fn
            {:public_comment, slug, thread, article_inner, comment_inner}, condition ->
              dynamic(
                [comment, gate_comment_binding: binding, gate_community: community],
                ^condition or
                  (community.slug == ^slug and comment.thread == ^thread and
                     binding.inner_id == ^article_inner and comment.inner_id == ^comment_inner)
              )
          end)

        matches_query =
          Comment
          |> CMS.Gate.scope(nil, :list, CommentScope.all_public())
          |> case do
            %Ecto.Query{} = query ->
              from(
                [comment, gate_comment_binding: binding, gate_community: community] in query,
                where: ^filter,
                where: as(:gate_comment_lifecycle).state == :visible,
                select: {comment, community.slug, binding.inner_id}
              )

            {:error, _reason} ->
              nil
          end

        matches = if matches_query, do: Repo.all(matches_query), else: []

        parents =
          matches
          |> Enum.group_by(fn {comment, _, _} -> comment.thread end, fn
            {_comment, slug, inner_id} -> {:public, slug, inner_id}
          end)
          |> Map.new(fn {thread, parent_locators} ->
            {thread, load_public_articles(thread, Enum.uniq(parent_locators))}
          end)

        Enum.reduce(matches, %{}, fn {comment, slug, article_inner}, acc ->
          key = {:public_comment, slug, comment.thread, article_inner, comment.inner_id}
          article = get_in(parents, [comment.thread, {:public, slug, article_inner}])

          if article do
            Map.put(acc, key, %{comment: comment, article: article, thread: comment.thread})
          else
            acc
          end
        end)
    end
  end

  defp comment_locator({:public_comment, _, _, _, _} = locator), do: locator

           article_id when not is_nil(article_id) <- Map.get(comment, info.foreign_key) do

        _ -> acc
      end
    end)
  end

  defp extract_links_from_text(text) do
    hrefs =
      @href_regex
      |> Regex.scan(text, capture: :all_but_first)
      |> Enum.map(fn captures ->
        Enum.find(captures, fn part -> is_binary(part) and part != "" end)
      end)
      |> Enum.reject(&is_nil/1)

    urls =
      @url_regex
      |> Regex.scan(text)
      |> Enum.map(&List.first/1)
      |> Enum.reject(&is_nil/1)

    Enum.uniq(hrefs ++ urls)
  end

  defp node_block_id(%{"id" => block_id}) when is_binary(block_id), do: block_id
  defp node_block_id(%{"_id" => block_id}) when is_binary(block_id), do: block_id
  defp node_block_id(_), do: "block-unknown"

  defp parse_mention_type(type) do
    case String.downcase(type) do
      "user" ->
        {:ok, :user}

      "comment" ->
        {:ok, :comment}

      normalized ->
        case Enum.find(@threads, &(Atom.to_string(&1) == normalized)) do
          nil -> {:error, ErrorCat.invalid_mention_type()}
          thread -> {:ok, thread}
        end
    end
  end

  defp parse_thread_slug(slug) when is_binary(slug) do
    normalized = String.downcase(slug)

    case Enum.find(@threads, fn thread -> Atom.to_string(thread) == normalized end) do
      nil -> {:error, ErrorCat.invalid_thread()}
      thread -> {:ok, thread}
    end
  end

  defp parse_thread_slug(_), do: {:error, ErrorCat.invalid_thread()}

  defp site_article_link?(url) do
    Enum.any?(@valid_article_prefix, &String.starts_with?(url, &1))
  end

  defp link_for_comment?(url) do
    with %{query: query} <- URI.parse(url) do
      not is_nil(query) and String.starts_with?(query, "comment_id=")
    end
  end

  defp hash_url(url) do
    :sha256
    |> :crypto.hash(url)
    |> Base.encode16(case: :lower)
  end
end
