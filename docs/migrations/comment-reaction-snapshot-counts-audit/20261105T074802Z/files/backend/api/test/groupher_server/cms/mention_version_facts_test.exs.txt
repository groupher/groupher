    refute :branch_id in Endpoint.__schema__(:fields)
    identity = %{type: :doc, article_id: Ecto.UUID.generate()}
    assert {:ok, first} = Store.ensure_endpoint_in_transaction(identity)

  @tag :stable_doc_identity
  test "Doc capture keeps one endpoint across source tokens and excludes stable self references" do
    {community, doc, _attrs, _user} = mock_article(:doc)
    endpoint = Repo.get_by!(Endpoint, type: :doc, article_id: doc.article_id)

    self_mention = [
      %{"type" => "mention", "key" => "doc:#{doc.article_id}", "children" => [%{"text" => ""}]}
    ]

    mentioned_at = DateTime.utc_now() |> DateTime.truncate(:second)

    for token <- ["#{doc.branch_id}:version:1", "#{doc.branch_id}:version:2"] do
      assert {:ok, {:ok, captured}} =
               Repo.transaction(fn ->
                 CaptureCurrentVersion.execute_in_transaction(
                   %{type: :doc, article_id: doc.article_id},
                   token,
                   self_mention,
                   community.id,
                   mentioned_at,
                   {:command, Ecto.UUID.generate()}
                 )
               end)

      assert captured.source_endpoint_id == endpoint.id
      assert captured.set.source_token == token
      refute Repo.exists?(from(o in Occurrence, where: o.set_id == ^captured.set.id))
    end
  end

  test "one source version has one set, including an empty set" do
    assert [%RevisionRow{id: 1, revision: 0}] = Repo.all(RevisionRow)

    source = endpoint!(:post, %{article_id: Ecto.UUID.generate()})
    attrs = %{source_endpoint_id: source.id, source_token: "revision:1", parser_version: 1}

    assert {:ok, set} = Repo.insert(MentionSet.changeset(%MentionSet{}, attrs))
    assert set.source_endpoint_id == source.id
    assert {:error, duplicate} = Repo.insert(MentionSet.changeset(%MentionSet{}, attrs))
    assert duplicate.errors[:source_endpoint_id]

    assert {:ok, newer_parser} =
             Repo.insert(MentionSet.changeset(%MentionSet{}, %{attrs | parser_version: 2}))

    assert newer_parser.id != set.id

    assert {:error, immutable} =
             set
             |> MentionSet.changeset(%{source_token: "revision:2"})
             |> Repo.update()

    assert immutable.errors[:base]
    assert Repo.aggregate(Occurrence, :count) == 0
  end

  test "parser upgrade keeps the old set unsynced until the new set projects" do
    source_identity = %{type: :post, article_id: Ecto.UUID.generate()}
    mentioned_at = DateTime.utc_now() |> DateTime.truncate(:second)

    save = fn parser_version ->
      Repo.transaction(fn ->
        Store.save_parsed_version_in_transaction(
          source_identity,
          "same-publication-token",
          parser_version,
          [],
          mentioned_at
        )
      end)
    end

    assert {:ok, {:ok, first}} = save.(1)
    assert {:ok, 0} = Store.project_set(first.source_endpoint_id, first.set.id)

    assert {:ok, {:ok, upgraded}} = save.(2)
    assert upgraded.set.id != first.set.id

    state = Repo.get!(ProjectionState, first.source_endpoint_id)
    assert state.desired_set_id == upgraded.set.id
    assert state.synced_set_id == first.set.id

    assert {:error, :mention_projection_set_not_desired} =
             Store.project_set(first.source_endpoint_id, first.set.id)

    assert {:ok, 0} = Store.project_set(first.source_endpoint_id, upgraded.set.id)

    state = Repo.get!(ProjectionState, first.source_endpoint_id)
    assert state.desired_set_id == upgraded.set.id
    assert state.synced_set_id == upgraded.set.id
  end

  test "initializer preserves a hidden Comment fragment and creates its auxiliary parent" do
    {community, parent, _attrs, user} = mock_article(:blog)

             CMS.Comments.create_comment(

    body =
      Jason.encode!([
        %{"type" => "p", "id" => "comment-fragment", "children" => [%{"text" => url}]}
      ])

    {:ok, %{article: source, draft: draft}} =
      CMS.Articles.create_stable_draft(
        community,
        :post,
        %{
          title: "Comment fragment source",
          digest: "Hidden Comment target",
          body_bag: mock_body_bag(body)
        },
        user
      )

    assert {:ok, _published} =
             CMS.Articles.publish(source.id, user,
               community: community,
               expected_draft_version: draft.version,
               expected_lifecycle_version: 1,
               command_id: Ecto.UUID.generate()
             )

    Repo.delete!(Repo.get!(CMS.Model.ArticlePublic, parent.article_id))
    token = maintenance_token("cms:mentions:initialize")

    assert {:ok, page} = InitializeCurrentSets.report_page(token, :articles, nil, 100)
    assert length(page["entries"]) == 1

    entry = hd(page["entries"])

    assert Enum.any?(entry["candidates"], fn candidate ->
             get_in(candidate, ["target_identity", "type"]) == "comment" and
               get_in(candidate, ["target_identity", "comment_id"]) == comment.id
           end)

    parent_endpoint_before = Repo.get_by!(Endpoint, type: :blog, article_id: parent.article_id)
    parent_state_before = Repo.get!(ProjectionState, parent_endpoint_before.id)

    assert {:ok, %{applied: 1}} = InitializeCurrentSets.apply_page(token, page)

    parent_endpoint = Repo.get_by!(Endpoint, type: :blog, article_id: parent.article_id)
    comment_endpoint = Repo.get_by!(Endpoint, type: :comment, comment_id: comment.id)
    source_endpoint = Repo.get_by!(Endpoint, type: :post, article_id: source.id)
    current_edge = Repo.get_by!(CurrentEdge, source_endpoint_id: source_endpoint.id)

    assert comment_endpoint.parent_endpoint_id == parent_endpoint.id
    assert current_edge.target_endpoint_id == comment_endpoint.id
    assert Repo.get!(ProjectionState, parent_endpoint.id) == parent_state_before
  end

  test "unresolved internal links inherit the endpoint and timestamp after moving blocks" do
    source_identity = %{type: :post, article_id: Ecto.UUID.generate()}
    target_id = Ecto.UUID.generate()
    source_time = DateTime.utc_now() |> DateTime.truncate(:second)
    url = "https://groupher.com/blog/#{target_id}"

    previous_mention = %{
      mentioned_scope: :internal,
      mentioned_type: :blog,
      mentioned_id: target_id,
      mention_case: :link,
      artiment: %{article_id: target_id},
      occurrence: %{
        block_id: "paragraph",
        path: [0],
        raw_url: url,
        normalized_from: "link"
      }
    }

    assert {:ok, previous_set} =
             Store.save_parsed_version(
               source_identity,
               "revision:previous",
               1,
               [previous_mention],
               source_time
             )

    source =
      Repo.get_by!(Endpoint, type: :post, article_id: source_identity.article_id)

    [previous_occurrence] =
      Repo.all(
        Ecto.Query.from(occurrence in Occurrence, where: occurrence.set_id == ^previous_set.id)
      )

    candidate = %{
      internal_mention_candidate?: true,
      type: :blog,
      value: target_id,
      occurrence: %{
        block_id: "paragraph",
        path: [3, 1],
        raw_url: url,
        normalized_from: "link"
      }
    }

    assert {:ok, [inherited]} =
             Store.inherit_unresolved_candidates(source_identity, [candidate], [])

    assert inherited.trusted_target_endpoint_id == previous_occurrence.target_endpoint_id

    assert {:ok, next_set} =
             Store.save_parsed_version(
               source_identity,
               "revision:next",
               1,
               [inherited],
               DateTime.add(source_time, 60, :second)
             )

    [next_occurrence] =
      Repo.all(
        Ecto.Query.from(occurrence in Occurrence, where: occurrence.set_id == ^next_set.id)
      )

    assert next_occurrence.target_endpoint_id == previous_occurrence.target_endpoint_id
    assert next_occurrence.mentioned_at == source_time
    assert next_occurrence.path == "3.1"
    assert next_occurrence.normalized_url == previous_occurrence.normalized_url
    assert next_occurrence.set_id != previous_occurrence.set_id
    assert Repo.get!(ProjectionState, source.id).desired_set_id == next_set.id

    assert {:ok, {:ok, :pass}} =
             Repo.transaction(fn ->
               Store.mark_endpoint_deleted_in_transaction(%{
                 type: :blog,
                 article_id: target_id
               })
             end)

    deleted_target = Repo.get!(Endpoint, previous_occurrence.target_endpoint_id)
    assert deleted_target.deleted_at

    body = [%{"type" => "p", "id" => "paragraph", "children" => [%{"text" => url}]}]

    assert {:ok, {:ok, captured}} =
             Repo.transaction(fn ->
               CaptureCurrentVersion.execute_in_transaction(
                 source_identity,
                 "revision:command-after-delete",
                 body,
                 nil,
                 DateTime.add(source_time, 120, :second),
                 {:command, Ecto.UUID.generate()}
               )
             end)

    [command_occurrence] =
      Repo.all(
        Ecto.Query.from(occurrence in Occurrence, where: occurrence.set_id == ^captured.set.id)
      )

    assert command_occurrence.target_endpoint_id == previous_occurrence.target_endpoint_id
    assert command_occurrence.mentioned_at == source_time
    assert Repo.get!(Endpoint, command_occurrence.target_endpoint_id).deleted_at
  end

  test "purging the main body preserves Doc identity while another branch still owns the Article" do
    {community, doc, _attrs, user} = mock_article(:doc)
    article = Repo.get!(CMS.Model.Article, doc.article_id)
    branch = Repo.get!(CMS.Model.DocBranch, doc.branch_id)
    endpoint = Repo.get_by!(Endpoint, type: :doc, article_id: article.id)

    assert {:ok, preview} =
             CMS.Docs.Branch.create_preview(community, %{slug: "retained-preview"}, user)

    Repo.insert!(
      CMS.Model.DocLifecycle.changeset(%CMS.Model.DocLifecycle{}, %{
        article_id: article.id,
        community_id: community.id,
        branch_id: preview.id,
        state: :draft,
        version: 1,
        changed_at: DateTime.utc_now() |> DateTime.truncate(:second)
      })
    )

    target = endpoint!(:blog, %{article_id: Ecto.UUID.generate()})
    set = set!(endpoint, "main-before-purge", 1)

    Repo.insert!(
      Occurrence.changeset(
        %Occurrence{},
        occurrence("retained", target, DateTime.utc_now() |> DateTime.truncate(:second))
        |> Map.put(:set_id, set.id)
      )
    )

    Repo.get!(ProjectionState, endpoint.id)
    |> ProjectionState.changeset(%{
      desired_set_id: set.id,
      desired_content_token: set.source_token
    })
    |> Repo.update!()

    assert {:ok, 1} = Store.project_set(endpoint.id, set.id)

    assert {:ok, comment_endpoint} =
             Store.ensure_endpoint_in_transaction(%{
               type: :comment,
               comment_id: 7_654_324,
               parent_endpoint_id: endpoint.id
             })

    incoming_set = set!(target, "incoming-before-purge", 1)

    Repo.insert!(
      Occurrence.changeset(
        %Occurrence{},
        occurrence("incoming", endpoint, DateTime.utc_now() |> DateTime.truncate(:second))
        |> Map.put(:set_id, incoming_set.id)
      )
    )

    Repo.insert!(
      ProjectionState.changeset(%ProjectionState{}, %{
        source_endpoint_id: target.id,
        desired_set_id: incoming_set.id,
        desired_content_token: incoming_set.source_token
      })
    )

    assert {:ok, 1} = Store.project_set(target.id, incoming_set.id)

    assert {:ok, {:ok, :done}} =
             Repo.transaction(fn ->
               {:ok, action} =
                 CMS.Docs.Trash.create_action(community, user, %{
                   root_type: :doc,
                   root_ref: article.id
                 })

               {:ok, item} =
                 CMS.Docs.Trash.attach(action, community, branch, article.id, user, audit: false)

               CMS.Docs.Trash.permanently_delete_loaded(item, article, branch, user, audit: false)
             end)

    assert Repo.get(CMS.Model.Article, article.id)
    assert is_nil(Repo.get!(Endpoint, endpoint.id).deleted_at)
    assert Repo.get(MentionSet, set.id)
    assert Repo.get!(Endpoint, comment_endpoint.id).parent_endpoint_id == endpoint.id
    assert is_nil(Repo.get!(Endpoint, comment_endpoint.id).deleted_at)
    assert Repo.exists?(from(edge in CurrentEdge, where: edge.target_endpoint_id == ^endpoint.id))
    refute Repo.exists?(from(edge in CurrentEdge, where: edge.source_endpoint_id == ^endpoint.id))

    assert {:ok, same} =
             Store.ensure_endpoint_in_transaction(%{type: :doc, article_id: article.id})

    assert same.id == endpoint.id

    {:ok, late_event} =
      Effects.enqueue_projection(endpoint.id, set.id, {:command, Ecto.UUID.generate()})

    assert {:ok, :pass} =
             CMSMentionSetProjector.perform(%Oban.Job{
               args: %{"event_id" => late_event.id},
               attempt: 1,
               max_attempts: 8
             })

    refute Repo.exists?(from(edge in CurrentEdge, where: edge.source_endpoint_id == ^endpoint.id))
  end

  test "saved-set rebuild requires its own maintenance scope and remains idempotent and silent" do
    {_community, source, _, _} = mock_article(:post)
    endpoint = Repo.get_by!(Endpoint, type: :post, article_id: source.article_id)
    command = CMS.ArtimentMentions.Commands.RebuildCurrentSet
    events_before = Repo.aggregate(OutboxEvent, :count)
    assert {:error, :mentions_maintenance_unauthorized} = command.execute(nil, endpoint.id)

    assert {:error, :mentions_maintenance_unauthorized} =
             command.execute(maintenance_token("cms:mentions:initialize"), endpoint.id)

    token = maintenance_token("cms:mentions:rebuild")
    assert {:ok, 0} = command.execute(token, endpoint.id)
    state = Repo.get!(ProjectionState, endpoint.id)
    assert state.desired_set_id == state.synced_set_id
    assert {:ok, 0} = command.execute(token, endpoint.id)
    assert Repo.aggregate(OutboxEvent, :count) == events_before
    assert Repo.aggregate(GroupherServer.Messaging.Model.Mention, :count) == 0
  end

  defp next_public_token(article_id) do
    public = Repo.get!(CMS.Model.ArticlePublic, article_id)

    updated =
      public
      |> Ecto.Changeset.change(publication_version: public.publication_version + 1)
      |> Repo.update!()

  end

  defp maintenance_token(scope) do
    verifier = GroupherServerWeb.ServiceAuth.Verifier
    previous = Application.get_env(:groupher_server, verifier)
    private_key = JOSE.JWK.generate_key({:rsa, 2048})
    public_key = JOSE.JWK.to_public(private_key)

    public_key_map =
      public_key
      |> JOSE.JWK.to_map()
      |> elem(1)
      |> Map.put("kid", "mentions-test-key")

    Application.put_env(:groupher_server, verifier,
      issuer: "https://auth.test",
      audiences: ["phoenix:cms-maintenance"],
      jwks: %{"keys" => [public_key_map]}
    )

    on_exit(fn ->
      if previous do
        Application.put_env(:groupher_server, verifier, previous)
      else
        Application.delete_env(:groupher_server, verifier)
      end
    end)

    now = System.system_time(:second)

    claims = %{
      "iss" => "https://auth.test",
      "aud" => "phoenix:cms-maintenance",
      "sub" => "service:mentions-initializer-test",
      "scope" => scope,
      "jti" => Ecto.UUID.generate(),
      "iat" => now,
      "nbf" => now,
      "exp" => now + 120
    }

    JOSE.JWT.sign(
      private_key,
      %{"alg" => "RS256", "kid" => "mentions-test-key", "typ" => "service_access+jwt"},
      claims
    )
    |> JOSE.JWS.compact()
    |> elem(1)
  end

    end
