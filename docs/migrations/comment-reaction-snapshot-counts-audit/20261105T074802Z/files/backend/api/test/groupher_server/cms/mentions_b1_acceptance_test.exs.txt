  @tag :b1_exit
  test "published User mentions deliver once through the actual projection event" do
    {community, _parent, _, author} = mock_article(:post)
    {:ok, recipient} = db_insert(:user)
    ast = mention_ast("user:#{recipient.login}")
    {source, publication} = publish_source(community, author, :post, ast)
    event = source_event(source.id)
    assert {:ok, :pass} = project(event)
    assert [%Mention{to_user_id: id} = mention] = Repo.all(Mention)
    assert id == recipient.id
    assert {:ok, _} = Messaging.mark_read(:mention, [mention.id], recipient)
    assert {:ok, :pass} = project(event)
    assert [%Mention{id: same_id, read: true}] = Repo.all(Mention)
    assert same_id == mention.id
    events_before = Repo.aggregate(Event, :count)
    assert {:ok, _} = CMS.Articles.publish(source.id, author, publication.opts)
    assert Repo.aggregate(Event, :count) == events_before
    assert Repo.aggregate(Mention, :count) == 1
  end

  @tag :b1_exit
  test "rebuild repairs a missing index from the immutable current set without reparsing" do
    {community, target, _, author} = mock_article(:blog)

    {source, _} =
      publish_source(community, author, :post, mention_ast("blog:#{target.article_id}"))

    assert {:ok, :pass} = project(source_event(source.id))
    endpoint = Repo.get_by!(Endpoint, type: :post, article_id: source.id)
    before = Repo.get!(ProjectionState, endpoint.id)
    [edge] = Repo.all(from(e in CurrentEdge, where: e.source_endpoint_id == ^endpoint.id))
    Repo.delete!(edge)
    Repo.delete!(Repo.get!(CMS.Model.ArticlePublic, target.article_id))
    assert {:ok, 1} = apply(Store, :rebuild_current_set, [endpoint.id])
    [rebuilt] = Repo.all(from(e in CurrentEdge, where: e.source_endpoint_id == ^endpoint.id))
    assert rebuilt.target_endpoint_id == edge.target_endpoint_id
    assert rebuilt.mentioned_at == edge.mentioned_at
    assert rebuilt.edge_key == edge.edge_key
    assert Repo.get!(ProjectionState, endpoint.id) == before
    assert {:ok, 0} = apply(Store, :rebuild_current_set, [endpoint.id])
    assert Repo.aggregate(Mention, :count) == 0
  end

  @tag :b1_exit
  test "all supported input syntaxes resolve facts, preserve external URLs, and reject wrong types at capture" do
    {community, target, _, user} = mock_article(:blog)
    {:ok, recipient} = db_insert(:user)

    {:ok, comment} =
      CMS.Comments.create_comment(
        community,
        :blog,
        article_inner_id(target, community),
        mock_comment("Parser fragment target"),
        user,
        Ecto.UUID.generate()
      )

    url =
      "#{CMS.ArtimentMentions.Config.site_host()}/#{community.slug}/blog/#{article_inner_id(target, community)}"

    for ast <- [
          mention_ast("blog:#{target.article_id}"),
          [
            %{
              "type" => "p",
              "children" => [
                %{"type" => "a", "url" => url, "children" => [%{"text" => "Link label"}]}
              ]
            }
          ],
          [%{"type" => "p", "children" => [%{"text" => ~s(<a href="#{url}">Anchor label</a>)}]}],
          [%{"type" => "p", "children" => [%{"text" => url}]}],
          [
            %{
              "type" => "a",
              "url" => "/#{community.slug}/blog/#{article_inner_id(target, community)}",
              "children" => [%{"text" => "Relative link"}]
            }
          ]
        ] do
      assert [%{mentioned_type: :blog, mentioned_id: id}] =
               Parser.parse(ast, community_id: community.id)

      assert id == target.article_id
      assert {:ok, [%{mentioned_id: ^id}]} = Parser.parse_for_initialization(ast)
    end

    for ast <- [
          mention_ast("comment:#{comment.id}"),
          [%{"type" => "p", "children" => [%{"text" => "#{url}#comment-#{comment.inner_id}"}]}],
          [
            %{
              "type" => "p",
              "children" => [
                %{
                  "text" =>
                    "#{CMS.ArtimentMentions.Config.site_host()}/blog/#{target.article_id}?comment_id=#{comment.id}"
                }
              ]
            }
          ]
        ] do
      assert [%{mentioned_type: :comment, mentioned_id: id}] =
               Parser.parse(ast, community_id: community.id)

      assert id == comment.id
      assert {:ok, [%{mentioned_id: ^id}]} = Parser.parse_for_initialization(ast)
    end

    assert [%{mentioned_type: :user, mentioned_id: recipient_id}] =
             Parser.parse(mention_ast("user:#{recipient.login}"))

    assert recipient_id == recipient.id

    for url <- [
          "https://external.example/guide",
          "https://customer.example/#{community.slug}/blog/1"
        ] do
      ast = [%{"type" => "p", "children" => [%{"text" => url}]}]
      assert [%{mentioned_type: :url, mentioned_url: ^url}] = Parser.parse(ast)

      assert {:ok, [%{mentioned_type: :url, mentioned_url: ^url}]} =
               Parser.parse_for_initialization(ast)
    end

    assert Parser.parse(mention_ast("user:user-does-not-exist")) == []
    assert Parser.parse(mention_ast("blog:#{Ecto.UUID.generate()}")) == []
    assert Parser.parse(mention_ast("unknown:resource")) == []

    assert {:error, :mention_target_type_mismatch} =
             Parser.parse_for_initialization(mention_ast("post:#{target.article_id}"))

    assert {:error, :mention_target_type_mismatch} =
             Repo.transaction(fn ->
               case CMS.ArtimentMentions.Commands.CaptureCurrentVersion.execute_in_transaction(
                      %{type: :post, article_id: Ecto.UUID.generate()},
                      "wrong-type",
                      mention_ast("post:#{target.article_id}"),
                      community.id,
                      DateTime.utc_now(:second),
                      {:command, Ecto.UUID.generate()}
                    ) do
                 {:error, reason} -> Repo.rollback(reason)
                 result -> result
               end
             end)
  end

  defp assert_normal_admission(:public, result, target_id) do
    assert [%{mentioned_id: ^target_id}] = result
  end

  defp assert_normal_admission(_hidden_state, result, _target_id), do: assert(result == [])

  defp change_comment_state(state, thread, parent, comment, community, author) do
    case state do
      :public ->
        :ok

      :comment_review ->
        Repo.insert!(%CMS.Model.CommentReviewBlock{
          comment_id: comment.id,
          type: :human_review_hold,
          source: :request,
          source_ref: Ecto.UUID.generate(),
          applied_at: DateTime.utc_now(:second)
        })

      :comment_deleted ->
        assert {:ok, _} = CMS.Comments.delete_comment(comment, author, Ecto.UUID.generate())

      :parent_review ->
        change_target_state(thread, :review_hidden, parent, community)

      :parent_deleted ->
        change_target_state(thread, :soft_deleted, parent, community)

      :community_hidden ->
        change_target_state(thread, :community_hidden, parent, community)
    end
  end

  defp change_target_state(_thread, :public, _target, _community), do: :ok

  defp change_target_state(:doc, :review_hidden, target, _community) do
    Repo.insert!(%CMS.Model.DocReviewBlock{
      article_id: target.article_id,
      branch_id: target.branch_id,
      type: :human_review_hold,
      source: :request,
      source_ref: Ecto.UUID.generate(),
      applied_at: DateTime.utc_now(:second)
    })
  end

  defp change_target_state(_thread, :review_hidden, target, _community) do
    Repo.insert!(%CMS.Model.ArticleReviewBlock{
      article_id: target.article_id,
      type: :human_review_hold,
      source: :request,
      source_ref: Ecto.UUID.generate(),
      applied_at: DateTime.utc_now(:second)
    })
  end

  defp change_target_state(:doc, :soft_deleted, target, _community) do
    assert {:ok, _} = CMS.Docs.Lifecycle.transition(target.article_id, target.branch_id, :deleted)
  end

  defp change_target_state(_thread, :soft_deleted, target, _community) do
    row = Repo.get_by!(CMS.Model.ArticleLifecycle, article_id: target.article_id)

    assert {:ok, {:ok, _}} =
             Repo.transaction(fn -> CMS.Articles.Lifecycle.transition(row, :deleted) end)
  end

  defp change_target_state(:doc, :no_public, target, _community) do
    Repo.delete_all(from(p in CMS.Model.DocPublic, where: p.article_id == ^target.article_id))
  end

  defp change_target_state(_thread, :no_public, target, _community) do
    Repo.delete!(Repo.get!(CMS.Model.ArticlePublic, target.article_id))
  end

  defp change_target_state(_thread, :community_hidden, _target, community) do
    case Repo.get_by(CMS.Model.CommunityLifecycle, community_id: community.id) do
      nil ->
        Repo.insert!(%CMS.Model.CommunityLifecycle{
          community_id: community.id,
          state: :archived,
          version: 1
        })

      row ->
        row |> Ecto.Changeset.change(state: :archived) |> Repo.update!()
    end
  end

  @tag :b1_exit
  test "Comment edits notify new recipients, preserve reads, and reclaim only superseded synced sets" do
    {community, parent, _, author} = mock_article(:blog)
    {:ok, recipient} = db_insert(:user)

    {:ok, comment} =
      CMS.Comments.create_comment(
        community,
        :blog,
        article_inner_id(parent, community),
        comment_bag(mention_ast("user:#{recipient.login}")),
        author,
        Ecto.UUID.generate()
      )

    endpoint = Repo.get_by!(Endpoint, comment_id: comment.id)
    original = Repo.get!(ProjectionState, endpoint.id)
    original_event = event_for_endpoint(endpoint.id, original.desired_set_id)
    assert {:ok, :pass} = project(original_event)
    [delivery] = Repo.all(Mention)
    assert {:ok, _} = Messaging.mark_read(:mention, [delivery.id], recipient)

    for ast <- [
          mention_ast("user:#{recipient.login}"),
          [%{"type" => "p", "children" => [%{"text" => "No references in this version"}]}],
          mention_ast("user:#{recipient.login}")
        ] do
      body = Repo.get_by!(CMS.Model.CommentBody, comment_id: comment.id)
      previous = Repo.get!(ProjectionState, endpoint.id)

      assert {:ok, _} =
               CMS.Comments.update_comment(
                 comment,
                 comment_bag(ast),
                 body.version,
                 author,
                 Ecto.UUID.generate()
               )

      next = Repo.get!(ProjectionState, endpoint.id)
      assert Repo.get(MentionSet, previous.desired_set_id)
      event = event_for_endpoint(endpoint.id, next.desired_set_id)
      assert {:ok, :pass} = project(event)
      refute Repo.get(MentionSet, previous.desired_set_id)
      assert Repo.get(MentionSet, next.desired_set_id)
      assert {:ok, :pass} = project(original_event)
    end

    assert Repo.aggregate(Mention, :count) == 2
    assert Repo.get!(Mention, delivery.id).read

    assert Repo.aggregate(
             from(set in MentionSet, where: set.source_endpoint_id == ^endpoint.id),
             :count
           ) == 1

    assert_initializer_is_silent(endpoint, mention_ast("user:#{recipient.login}"))
  end

  @tag :b1_time_identity
  test "two URL spellings of one internal relationship inherit one mentioned_at" do
    {community, target, _, _} = mock_article(:blog)
    identity = %{type: :post, article_id: Ecto.UUID.generate()}
    first_at = DateTime.add(DateTime.utc_now(:second), -3600, :second)
    host = CMS.ArtimentMentions.Config.site_host()
    legacy = "#{host}/blog/#{target.article_id}"
    public = "#{host}/#{community.slug}/blog/#{article_inner_id(target, community)}"

    capture = fn token, urls, at ->
      ast = Enum.map(urls, fn url -> %{"type" => "p", "children" => [%{"text" => url}]} end)

      Repo.transaction(fn ->
        CMS.ArtimentMentions.Commands.CaptureCurrentVersion.execute_in_transaction(
          identity,
          token,
          ast,
          community.id,
          at,
          {:command, Ecto.UUID.generate()}
        )
      end)
    end

    assert {:ok, {:ok, first}} = capture.("urls:1", [legacy], first_at)

    assert {:ok, {:ok, second}} =
             capture.("urls:2", [legacy, public], DateTime.add(first_at, 60, :second))

    times =
      from(o in Occurrence, where: o.set_id == ^second.set.id, select: o.mentioned_at)
      |> Repo.all()

    assert length(times) == 2
    assert Enum.uniq(times) == [first_at]
    assert {:ok, 1} = Store.project_set(second.source_endpoint_id, second.set.id)

    assert Repo.get_by!(CurrentEdge, source_endpoint_id: first.source_endpoint_id).mentioned_at ==
             first_at
  end

  test "Doc Trash and restore preserve incoming node identity, timestamp and read notification" do
    {community, doc, _, author} = mock_article(:doc)
    {:ok, recipient} = db_insert(:user)

    {source, _} =
      publish_source(
        community,
        author,
        :post,
        mention_ast("doc:#{doc.article_id}") ++ mention_ast("user:#{recipient.login}")
      )

    assert {:ok, :pass} = project(source_event(source.id))
    [delivery] = Repo.all(Mention)
    assert {:ok, _} = Messaging.mark_read(:mention, [delivery.id], recipient)
    doc_endpoint = Repo.get_by!(Endpoint, type: :doc, article_id: doc.article_id)
    edge = Repo.get_by!(CurrentEdge, target_endpoint_id: doc_endpoint.id)
    focus = %{type: :post, article_inner_id: article_inner_id(source, community)}
    {:ok, page} = CMS.mentions_map_page(community.slug, focus, [%{node: focus}], 20)
    node = Enum.find(page.nodes, &(&1.type == :doc))
    assert node.status == :readable
    branch = Repo.get!(CMS.Model.DocBranch, doc.branch_id)

    assert {:ok, item} =
             Repo.transaction(fn ->
               {:ok, action} =
                 CMS.Docs.Trash.create_action(community, author, %{
                   root_type: :doc,
                   root_ref: doc.article_id
                 })

               {:ok, item} =
                 CMS.Docs.Trash.attach(action, community, branch, doc.article_id, author,
                   audit: false
                 )

               item
             end)

    {:ok, hidden_page} = CMS.mentions_map_page(community.slug, focus, [%{node: focus}], 20)
    hidden = Enum.find(hidden_page.nodes, &(&1.type == :doc))
    assert hidden.status == :unavailable
    assert is_nil(hidden.title)
    assert is_nil(hidden.url)
    assert hidden.id == node.id

      Enum.with_index(target_ids, fn id, index ->
        %{
          mentioned_scope: :internal,

          mention_case: :link,

  @tag :b1_doc_comment_source
  test "Doc Comment source delivers once, reuses its stable parent and initializes silently" do
    {community, parent, _, author} = mock_article(:doc)
    {:ok, recipient} = db_insert(:user)
    ast = mention_ast("user:#{recipient.login}")
    {:ok, comment} = CMS.Comments.create_comment(community, :doc, article_inner_id(parent, community),
      comment_bag(ast), author, Ecto.UUID.generate())
    endpoint = Repo.get_by!(Endpoint, comment_id: comment.id)
    parent_endpoint = Repo.get_by!(Endpoint, type: :doc, article_id: parent.article_id)
    assert endpoint.parent_endpoint_id == parent_endpoint.id
    state = Repo.get!(ProjectionState, endpoint.id)
    event = event_for_endpoint(endpoint.id, state.desired_set_id)
    assert {:ok, :pass} = project(event)
    assert {:ok, :pass} = project(event)
    assert [%Mention{comment_id: id, to_user_id: recipient_id}] = Repo.all(Mention)
    assert id == comment.id
    assert recipient_id == recipient.id
    assert_initializer_is_silent(endpoint, ast)
  end

  defp assert_initializer_is_silent(endpoint, ast) do
    category =
      case endpoint.type do
        :doc -> :docs
        :comment -> :comments
        _ -> :articles
      end

    {:ok, sources} = CMS.ArtimentMentions.InitializeSourceScan.page(category, nil, 100)

    source =
      Enum.find(sources, fn source ->
        if endpoint.type == :comment do
          source.comment_id == endpoint.comment_id
        else
          source.article_id == endpoint.article_id
        end
      end)

    assert source
    before_token = CMS.ArtimentMentions.CurrentSource.load(endpoint)
    before_mentions = Repo.all(Mention)
    before_events = Repo.aggregate(Event, :count)
    Repo.delete_all(from(e in CurrentEdge, where: e.source_endpoint_id == ^endpoint.id))
    Repo.delete_all(from(s in ProjectionState, where: s.source_endpoint_id == ^endpoint.id))
    Repo.delete_all(from(s in MentionSet, where: s.source_endpoint_id == ^endpoint.id))
    assert {:ok, first} = CMS.ArtimentMentions.InitializationWorkflow.apply_source(source, ast)
    assert {:ok, second} = CMS.ArtimentMentions.InitializationWorkflow.apply_source(source, ast)
    assert first.set.id == second.set.id
    state = Repo.get!(ProjectionState, endpoint.id)
    assert state.desired_set_id == state.synced_set_id
    assert Repo.all(Mention) == before_mentions
    assert Repo.aggregate(Event, :count) == before_events
    assert CMS.ArtimentMentions.CurrentSource.load(endpoint) == before_token
  end
