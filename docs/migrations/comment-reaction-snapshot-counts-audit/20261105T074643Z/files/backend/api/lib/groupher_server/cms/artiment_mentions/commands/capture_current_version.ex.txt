defmodule GroupherServer.CMS.ArtimentMentions.Commands.CaptureCurrentVersion do
  @moduledoc """
  Saves the selected public body's Mention facts in its owning Command transaction.

      selected public body -> Parser -> immutable set + desired watermark -> Outbox projector
  """

  alias GroupherServer.CMS.ArtimentMentions.{Effects, Parser, Store}
  alias GroupherServer.CMS.ArtimentMentions.Model.ProjectionState
  alias GroupherServer.Repo

  @parser_version 1

  @doc "Captures one canonical version and dispatches projection within the caller transaction."
  def execute_in_transaction(
        source_identity,
        source_token,
        ast,
        community_id,
        mentioned_at,
        outbox_identity
      )
      when is_map(source_identity) and is_binary(source_token) and is_list(ast) and
             (is_nil(community_id) or is_integer(community_id)) and
             is_struct(mentioned_at, DateTime) and is_tuple(outbox_identity) do
    execute_in_transaction(
      source_identity,
      source_token,
      ast,
      community_id,
      mentioned_at,
      outbox_identity,
      []
    )
  end

  def execute_in_transaction(_, _, _, _, _, _), do: {:error, :invalid_mention_capture}

  def execute_in_transaction(
        source_identity,
        source_token,
        ast,
        community_id,
        mentioned_at,
        outbox_identity,
        opts
      )
      when is_map(source_identity) and is_binary(source_token) and is_list(ast) and
             (is_nil(community_id) or is_integer(community_id)) and
             is_struct(mentioned_at, DateTime) and is_tuple(outbox_identity) and
             is_list(opts) do
    with true <- Repo.in_transaction?(),
         candidates <- Parser.candidates(ast),
         {:ok, :pass} <- Parser.validate_url_lengths(ast),
         {:ok, :pass} <- Parser.validate_identity_types(ast),
         {:ok, parsed_mentions} <- parse_mentions(ast, community_id, opts),
         resolved_mentions <- Enum.reject(parsed_mentions, &self_reference?(source_identity, &1)),
         {:ok, mentions} <-
           bind_prior_targets(
             Keyword.get(opts, :binding_mode),
             source_identity,
             candidates,
             resolved_mentions
           ),
         {:ok, saved} <-
           Store.save_parsed_version_in_transaction(
             source_identity,
             source_token,
             @parser_version,
             mentions,
             mentioned_at
           ),
         {:ok, dispatch_result} <-
           dispatch_projection(saved, outbox_identity, opts, source_identity.type) do
      {:ok, Map.put(saved, :dispatch_result, dispatch_result)}
    else
      false -> {:error, :mention_capture_requires_transaction}
      {:error, _reason} = error -> error
    end
  end

  def execute_in_transaction(_, _, _, _, _, _, _), do: {:error, :invalid_mention_capture}

  defp dispatch_projection(saved, outbox_identity, opts, source_type) do
    case Keyword.get(opts, :binding_mode) do
      :initialize_current ->
        case Repo.get(ProjectionState, saved.source_endpoint_id) do
          %ProjectionState{desired_set_id: set_id, synced_set_id: set_id}
          when set_id == saved.set.id ->
            {:ok, :unchanged}

          _ ->
            case Store.project_set(saved.source_endpoint_id, saved.set.id) do
              {:ok, _count} -> {:ok, :projected}
              {:error, _reason} = error -> error
            end
        end

      nil ->
        # Ticket inbox routing is outside the existing notification contract.
        recipients = if source_type == :ticket, do: [], else: saved.new_user_ids

        Effects.enqueue_projection(
          saved.source_endpoint_id,
          saved.set.id,
          outbox_identity,
          recipients
        )

      _ ->
        {:error, :invalid_mention_binding_mode}
    end
  end

  defp parse_mentions(ast, community_id, opts) do
    case Keyword.get(opts, :binding_mode) do
      nil ->
        {:ok, Parser.parse(ast, community_id: community_id)}

      :initialize_current ->
        with {:ok, :pass} <- Parser.validate_url_lengths(ast) do
          Parser.parse_for_initialization(ast)
        end

      _ ->
        {:error, :invalid_mention_binding_mode}
    end
  end

  defp bind_prior_targets(:initialize_current, _source_identity, _candidates, mentions) do
    {:ok, mentions}
  end

  defp bind_prior_targets(nil, source_identity, candidates, mentions) do
    Store.inherit_unresolved_candidates(source_identity, candidates, mentions)
  end

  defp self_reference?(
         %{type: type, article_id: id},
         %{mentioned_type: type, mentioned_id: id}
       )
       when type in [:post, :blog, :changelog, :doc, :ticket] do
    true
  end

  defp self_reference?(
         %{type: :comment, comment_id: id},
         %{mentioned_type: :comment, mentioned_id: id}
       ) do
    true
  end

  defp self_reference?(_, _), do: false
end
