defmodule GroupherServer.CMS.ArtimentMentions.Commands.RebuildCurrentSet do
  @moduledoc """
  Rebuilds one saved current Mention set under a dedicated maintenance capability.

      maintenance JWT -> exact rebuild scope -> canonical token -> saved-set projection

  This command neither reparses content nor publishes or notifies. The initializer
  capability cannot invoke it. It is an internal maintenance entry, not GraphQL.
  """
  alias GroupherServer.CMS.ArtimentMentions.{MaintenanceAuthorization, Store}

  def execute(token, source_endpoint_id) do
    with {:ok, _principal} <- MaintenanceAuthorization.verify(token, :rebuild),
         {:ok, id} <- Ecto.UUID.cast(source_endpoint_id) do
      Store.rebuild_current_set(id)
    else
      :error -> {:error, :invalid_mention_projection}
      {:error, reason} -> {:error, reason}
    end
  end
end
