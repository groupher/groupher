defmodule GroupherServer.CMS.ArtimentMentions.Commands.InitializeCurrentSets do
  @moduledoc """
  Reports and applies bounded pages of current canonical sources without publishing.

      maintenance capability -> frozen page report -> re-read and compare -> inline projection
  """

  alias GroupherServer.CMS

  alias CMS.ArtimentMentions.{
    InitializeSourceScan,
    MaintenanceAuthorization,
    Parser
  }

  alias CMS.ArtimentMentions.InitializationWorkflow
  alias CMS.Artiment.PlateJSON

  @parser_version 1
  @max_page_size 100

  @doc "Builds one manifest page from the current canonical source selections."
  def report_page(token, category, cursor \\ nil, limit \\ @max_page_size) do
    with {:ok, _principal} <- MaintenanceAuthorization.verify(token),
         {:ok, sources} <- InitializeSourceScan.page(category, cursor, limit),
         {:ok, entries} <- build_entries(sources) do
      {:ok,
       %{
         "schema_version" => 2,
         "parser_version" => @parser_version,
         "category" => Atom.to_string(category),
         "cursor" => encode_cursor(cursor),
         "limit" => limit,
         "entries" => entries,
         "next_cursor" => next_cursor(category, sources, limit)
       }}
    else
      {:error, _reason} = error -> error
    end
  end

  @doc "Preflights and applies a report page after re-reading its exact source rows."
  def apply_page(token, manifest_page) when is_map(manifest_page) do
    with {:ok, _principal} <- MaintenanceAuthorization.verify(token),
         {:ok, category} <- manifest_category(manifest_page),
         {:ok, cursor} <- manifest_cursor(manifest_page),
         {:ok, limit} <- manifest_limit(manifest_page),
         true <- fetch(manifest_page, :schema_version) == 2,
         true <- fetch(manifest_page, :parser_version) == @parser_version,
         {:ok, sources} <- InitializeSourceScan.page(category, cursor, limit),
         {:ok, current_entries} <- build_entries(sources),
         true <-
           canonical_entries(fetch(manifest_page, :entries)) ==
             canonical_entries(current_entries),
         {:ok, results} <- apply_sources(sources, current_entries) do
      {:ok,
       %{
         applied: Enum.count(results, &(&1.status == :applied)),
         unchanged: Enum.count(results, &(&1.status == :unchanged)),
         results: results,
         next_cursor: next_cursor(category, sources, limit)
       }}
    else
      false -> {:error, :mention_initialization_manifest_mismatch}
      {:error, _reason} = error -> error
    end
  end

  def apply_page(_, _), do: {:error, :invalid_mention_initialization_manifest}

  @doc "Checks authorization and source fingerprints without writing."
  def preflight_page(token, manifest_page) when is_map(manifest_page) do
    with {:ok, _principal} <- MaintenanceAuthorization.verify(token),
         {:ok, category} <- manifest_category(manifest_page),
         {:ok, cursor} <- manifest_cursor(manifest_page),
         {:ok, limit} <- manifest_limit(manifest_page),
         true <- fetch(manifest_page, :schema_version) == 2,
         true <- fetch(manifest_page, :parser_version) == @parser_version,
         {:ok, current_page} <- report_page(token, category, cursor, limit),
         true <-
           canonical_entries(fetch(manifest_page, :entries)) ==
             canonical_entries(current_page["entries"]) do
      {:ok, %{source_count: length(current_page["entries"])}}
    else
      false -> {:error, :mention_initialization_manifest_mismatch}
      {:error, _reason} = error -> error
    end
  end

  def preflight_page(_, _), do: {:error, :invalid_mention_initialization_manifest}

  defp build_entries(sources) do
    Enum.reduce_while(sources, {:ok, []}, fn source, {:ok, entries} ->
      case source_entry(source) do
        {:ok, entry} ->
          {:cont, {:ok, [entry | entries]}}

        {:error, reason} ->
          {:halt,
           {:error, {:mention_initialization_source_invalid, source_identity(source), reason}}}
      end
    end)
    |> case do
      {:ok, entries} -> {:ok, Enum.reverse(entries)}
      {:error, _reason} = error -> error
    end
  end

  defp source_entry(source) do
    with true <- is_binary(source.body_json) and is_binary(source.body_hash),
         %DateTime{} = mentioned_at <- source.mentioned_at,
         {:ok, ast} <- PlateJSON.decode(source.body_json),
         {:ok, parsed} <- Parser.parse_for_initialization(ast) do
      parsed = Enum.reject(parsed, &self_reference?(source_identity(source), &1))
      candidates = manifest_candidates(ast, parsed, source_identity(source))

      entry = %{
        "identity" => serialize_identity(source_identity(source)),
        "canonical_token" => source.source_token,
        "body_hash" => source.body_hash,
        "mentioned_at" => DateTime.to_iso8601(mentioned_at),
        "candidates" => candidates,
        "source_fingerprint" => fingerprint(source, candidates)
      }

      {:ok, entry}
    else
      false -> {:error, :canonical_body_missing}
      nil -> {:error, :source_publication_time_missing}
      {:error, reason} -> {:error, reason}
      _ -> {:error, :invalid_canonical_source}
    end
  end

  defp manifest_candidates(ast, parsed, source_identity) do
    bound = Map.new(parsed, &{occurrence_key(&1.occurrence), &1})

    ast
    |> Parser.candidates()
    |> Enum.reject(&self_candidate?(source_identity, &1))
    |> Enum.map(fn candidate ->
      key = occurrence_key(candidate.occurrence)

      case Map.get(bound, key) do
        %{mentioned_scope: :external} = mention ->
          %{
            "occurrence_key" => key,
            "resolution" => "bound",
            "target_identity" => %{
              "type" => "url",
              "url" => mention.mentioned_url,
              "url_hash" => mention.mentioned_url_hash
            }
          }

        %{mentioned_type: :user, mentioned_id: id} ->
          %{
            "occurrence_key" => key,
            "resolution" => "bound",
            "target_identity" => %{"type" => "user", "id" => id}
          }

        %{mentioned_type: :comment, mentioned_id: id} = mention ->
          parent = Map.get(mention, :parent_article)

          %{
            "occurrence_key" => key,
            "resolution" => "bound",
            "target_identity" => %{"type" => "comment", "comment_id" => id},
            "parent_identity" => serialize_identity(parent_identity(parent))
          }

        %{mentioned_type: type, mentioned_id: id} ->
          %{
            "occurrence_key" => key,
            "resolution" => "bound",
            "target_identity" => %{
              "type" => Atom.to_string(type),
              "article_id" => id
            }
          }

        nil ->
          unresolved_candidate(candidate, key)
      end
    end)
  end

  defp unresolved_candidate(candidate, key) do
    resolution =
      cond do
        Map.get(candidate, :internal_mention_candidate?) and
            get_in(candidate, [:occurrence, :normalized_from]) == "link" ->
          "ordinary_link"

        Map.get(candidate, :internal_mention_candidate?) ->
          "ordinary_text"

        true ->
          "ordinary_link"
      end

    %{
      "occurrence_key" => key,
      "resolution" => resolution,
      "reason" => "target_identity_unavailable"
    }
  end

  defp apply_sources(sources, entries) do
    by_identity = Map.new(entries, &{fetch(&1, :identity), &1})

    Enum.reduce_while(sources, {:ok, []}, fn source, {:ok, results} ->
      identity = serialize_identity(source_identity(source))
      entry = Map.get(by_identity, identity)

      case apply_source(source, entry) do
        {:ok, result} ->
          {:cont, {:ok, [result | results]}}

        {:error, reason} ->
          {:halt, {:error, {:mention_initialization_apply_failed, identity, reason}}}
      end
    end)
    |> case do
      {:ok, results} -> {:ok, Enum.reverse(results)}
      {:error, _reason} = error -> error
    end
  end

  defp apply_source(source, entry) when is_map(entry) do
    with {:ok, ast} <- PlateJSON.decode(source.body_json),
         {:ok, %{set: set, dispatch_result: dispatch_result}} <-
           InitializationWorkflow.apply_source(source, ast) do
      status = if dispatch_result == :unchanged, do: :unchanged, else: :applied

      {:ok,
       %{identity: serialize_identity(source_identity(source)), set_id: set.id, status: status}}
    else
      {:error, reason} -> {:error, reason}
    end
  end

  defp apply_source(_source, _entry),
    do: {:error, :mention_initialization_source_missing_from_manifest}

  defp source_identity(%{type: :comment, comment_id: id, parent: parent}) do
    %{
      type: :comment,
      comment_id: id,
      parent: parent_identity(parent)
    }
  end

  defp source_identity(%{type: type, article_id: id}), do: %{type: type, article_id: id}

  defp parent_identity(nil), do: nil

  defp parent_identity(parent) do
    %{
      type: field(parent, :thread) || field(parent, :type),
      article_id: field(parent, :article_id)
    }
  end

  defp serialize_identity(nil), do: nil

  defp serialize_identity(identity) do
    serialize_value(identity)
  end

  defp serialize_value(map) when is_map(map) do
    map
    |> Enum.reject(fn {_key, value} -> is_nil(value) end)
    |> Map.new(fn {key, value} -> {to_string(key), serialize_value(value)} end)
  end

  defp serialize_value(value) when is_atom(value), do: Atom.to_string(value)
  defp serialize_value(list) when is_list(list), do: Enum.map(list, &serialize_value/1)
  defp serialize_value(value), do: value

  defp field(map, key) when is_map(map), do: Map.get(map, key)
  defp field(_, _), do: nil

  defp occurrence_key(occurrence) do
    block_id = Map.get(occurrence, :block_id) || "block"
    path = occurrence |> Map.get(:path, []) |> Enum.join(".")
    "#{block_id}:#{path}"
  end

  defp self_reference?(
         %{type: type, article_id: id},
         %{mentioned_type: type, mentioned_id: id}
       )
       when type in [:post, :blog, :changelog, :doc, :ticket],
       do: true

  defp self_reference?(
         %{type: :comment, comment_id: id},
         %{mentioned_type: :comment, mentioned_id: id}
       ),
       do: true

  defp self_reference?(_, _), do: false

  defp self_candidate?(%{type: :comment, comment_id: id}, %{type: :comment, value: value}),
    do: to_string(id) == to_string(value)

  defp self_candidate?(%{type: type, article_id: id}, %{type: type, value: value}),
    do: to_string(id) == to_string(value)

  defp self_candidate?(_, _), do: false

  defp fingerprint(source, candidates) do
    payload = %{
      identity: serialize_identity(source_identity(source)),
      token: source.source_token,
      body_hash: source.body_hash,
      mentioned_at: DateTime.to_iso8601(source.mentioned_at),
      candidates: candidates
    }

    :crypto.hash(:sha256, Jason.encode!(payload)) |> Base.encode16(case: :lower)
  end

  defp next_cursor(_category, sources, limit) when length(sources) < limit, do: nil

  defp next_cursor(:articles, sources, _limit),
    do: sources |> List.last() |> Map.fetch!(:article_id)

  defp next_cursor(:docs, sources, _limit) do
    source = List.last(sources)
    [source.article_id, source.branch_id]
  end

  defp next_cursor(:comments, sources, _limit),
    do: sources |> List.last() |> Map.fetch!(:comment_id)

  defp encode_cursor(nil), do: nil
  defp encode_cursor({article_id, branch_id}), do: [article_id, branch_id]
  defp encode_cursor(cursor), do: cursor

  defp manifest_category(page) do
    case fetch(page, :category) do
      "articles" -> {:ok, :articles}
      "docs" -> {:ok, :docs}
      "comments" -> {:ok, :comments}
      _ -> {:error, :invalid_mention_initialization_category}
    end
  end

  defp manifest_cursor(page) do
    case fetch(page, :cursor) do
      nil ->
        {:ok, nil}

      [article_id, branch_id] when is_binary(article_id) and is_integer(branch_id) ->
        {:ok, {article_id, branch_id}}

      cursor when is_binary(cursor) or is_integer(cursor) ->
        {:ok, cursor}

      _ ->
        {:error, :invalid_mention_initialization_cursor}
    end
  end

  defp manifest_limit(page) do
    case fetch(page, :limit) do
      limit when is_integer(limit) and limit in 1..@max_page_size -> {:ok, limit}
      _ -> {:error, :invalid_mention_initialization_page_size}
    end
  end

  defp fetch(map, key) do
    Map.get(map, key, Map.get(map, Atom.to_string(key)))
  end

  defp canonical_entries(entries) when is_list(entries) do
    entries
    |> Enum.map(&canonical_term/1)
    |> Enum.sort_by(&Jason.encode!/1)
  end

  defp canonical_entries(_), do: nil

  defp canonical_term(map) when is_map(map) do
    map
    |> Enum.map(fn {key, value} -> {to_string(key), canonical_term(value)} end)
    |> Enum.sort_by(&elem(&1, 0))
    |> Map.new()
  end

  defp canonical_term(list) when is_list(list), do: Enum.map(list, &canonical_term/1)
  defp canonical_term(value), do: value
end
