defmodule GroupherServer.CMS.ArtimentMentions.MentionsMap.Query do
  @moduledoc """
  Reads a bounded page of current Mention edges for a Community and frontier batch.

      Community + locators -> revision fence -> current-edge windows -> public statuses
  """

  import Ecto.Query

  alias GroupherServer.{CMS, Repo}
  alias CMS.ArtimentMentions.{Opaque, MentionsMap}
  alias MentionsMap.{Budget, Graph, NodeStatus}
  alias CMS.ArtimentMentions.MentionsGraphRevision, as: Revision
  alias CMS.ArtimentMentions.Model.{CurrentEdge, Endpoint, MentionSet, ProjectionState}
  alias CMS.Gate.Context.Scope.Article, as: ArticleScope
  alias CMS.Gate.Context.Scope.Comment, as: CommentScope
  alias CMS.Gate.Context.Scope.Doc, as: DocScope
  alias CMS.Gate.Context.Scope.Ticket, as: TicketScope
  alias CMS.Model.{Article, Comment, CommentBody, Community, TicketState}
  alias GroupherServer.Accounts.Model.User
  alias GroupherServer.Pagi.Params
  alias Helper.ORM.Pagi.Page

  @max_frontiers 10
  @max_first 20

  @doc "Returns one bounded graph page and rejects pages changed during the read."
  def page(community_slug, focus, frontiers, first)
      when is_binary(community_slug) and is_map(focus) and is_list(frontiers) and
             length(frontiers) in 1..@max_frontiers and is_integer(first) and
             first in 1..@max_first do
    budget = Budget.new()

    transaction(budget, fn ->
      with {:ok, revision} <- Revision.current(budget),
           {:ok, community} <- community(community_slug, budget),
           {:ok, nodes} <-
             NodeStatus.resolve_locators(
               community,
               [focus | Enum.map(frontiers, &Map.get(&1, :node))],
               budget
             ),
           [focus_node | frontier_nodes] <- nodes,
           {:ok, :pass} <- validate_focus(focus_node),
           {:ok, :pass} <- validate_frontiers(focus_node, frontier_nodes),
           {:ok, edge_pages} <-
             load_frontier_edges(community, revision, frontier_nodes, frontiers, first, budget),
           {:ok, endpoint_nodes} <- load_edge_node_statuses(community, edge_pages, budget),
           {:ok, enriched_pages} <-
             attach_node_statuses(
               community,
               revision,
               first,
               edge_pages,
               frontier_nodes,
               endpoint_nodes
             ),
           graph_frontiers <-
             Enum.zip(frontiers, frontier_nodes)
             |> Enum.map(fn {input, node} -> %{input: input, node: node} end),
           {:ok, graph} <- Graph.build(community, focus_node, graph_frontiers, enriched_pages),
           {:ok, projection_pending} <-
             projection_pending(community, [focus_node | frontier_nodes], budget),
           {:ok, :pass} <- unchanged_revision(revision, budget) do
        {:ok, Map.put(graph, :projection_pending, projection_pending)}
      end
    end)
  end

  def page(_, _, _, _), do: {:error, CMS.ErrorCat.mentions_map_invalid_request()}

  @doc "Returns public Comment locators that have current edges or a pending source set."
  def article_comments(community_slug, focus, after_cursor, first)
      when is_binary(community_slug) and is_map(focus) and is_integer(first) and
             first in 1..@max_first do
    budget = Budget.new()

    transaction(budget, fn ->
      with {:ok, revision} <- Revision.current(budget),
           {:ok, community} <- community(community_slug, budget),
           {:ok, after_id} <- parse_comment_cursor(after_cursor),
           {:ok, [focus_node]} <- NodeStatus.resolve_locators(community, [focus], budget),
           {:ok, :pass} <- validate_article_comment_focus(focus_node),
           article_id <- elem(focus_node.identity, 1),
           rows <-
             comment_seed_rows(community, article_id, after_id, first + 1, budget),
           selected <- Enum.take(rows, first),
           {:ok, projection_pending} <- projection_pending(community, [focus_node], budget),
           {:ok, :pass} <- unchanged_revision(revision, budget) do
        {:ok,
         %{
           comments:
             Enum.map(selected, fn row ->
               %{
                 type: :comment,
                 article_inner_id: row.article_inner_id,
                 comment_inner_id: row.comment_inner_id
               }
             end),
           has_next_page: length(rows) > first,
           end_cursor:
             if(length(rows) > first and selected != [],
               do: to_string(List.last(selected).comment_inner_id)
             ),
           projection_pending: projection_pending
         }}
      else
        {:ok, _} -> {:error, CMS.ErrorCat.mentions_map_not_found()}
        {:error, _reason} = error -> error
      end
    end)
  end

  def article_comments(_, _, _, _), do: {:error, CMS.ErrorCat.mentions_map_invalid_request()}

  def revision(community_slug, focus) do
    budget = Budget.new()

    transaction(budget, fn ->
      with {:ok, community} <- community(community_slug, budget),
           {:ok, [focus_node]} <- NodeStatus.resolve_locators(community, [focus], budget),
           {:ok, :pass} <- validate_focus(focus_node),
           {:ok, revision} <- Revision.current(budget) do
        {:ok, Opaque.id(:revision, community.id, revision)}
      else
        {:error, _reason} = error -> error
        _ -> {:error, CMS.ErrorCat.mentions_map_not_found()}
      end
    end)
  end

  @doc "Returns one bounded, Community-scoped list of incoming or outgoing current Mention edges."
  def one_hop(community_slug, direction, root_input, pagi)
      when is_binary(community_slug) and direction in [:outgoing, :incoming] and
             is_map(root_input) do
    budget = Budget.new()

    transaction(budget, fn ->
      with {:ok, revision} <- Revision.current(budget),
           {:ok, community} <- community(community_slug, budget),
           {:ok, page_request} <- normalize_list_page(pagi),
           {:ok, root} <- resolve_list_root(community, direction, root_input, budget),
           {:ok, page_rows, page_info} <-
             list_edge_page(community, direction, root, revision, page_request, budget),
           {:ok, entries} <- project_list_entries(community, direction, root, page_rows, budget),
           {:ok, projection_pending} <- projection_pending(community, [], budget),
           {:ok, :pass} <- unchanged_revision(revision, budget) do
        {:ok, %{entries: entries, page_info: page_info, projection_pending: projection_pending}}
      end
    end)
  end

  def one_hop(_, _, _, _), do: {:error, CMS.ErrorCat.mentions_map_invalid_request()}

  defp normalize_list_page(nil), do: {:ok, %Params.Cursor{first: @max_first, after: nil}}

  defp normalize_list_page(%Params.Cursor{first: first} = page)
       when is_integer(first) and first in 1..@max_first,
       do: {:ok, page}

  defp normalize_list_page(%Params.Offset{page: page, size: size} = request)
       when is_integer(page) and page in 1..10_000 and is_integer(size) and size in 1..@max_first,
       do: {:ok, request}

  defp normalize_list_page(_), do: {:error, CMS.ErrorCat.mentions_map_invalid_request()}

  defp resolve_list_root(_community, direction, %{type: :user, user_login: login}, budget)
       when direction == :incoming and is_binary(login) do
    case Budget.run!(budget, fn timeout -> Repo.get_by(User, [login: login], timeout: timeout) end) do
      %User{} = user ->
        {:ok, %{identity: {:user, user.id}, user: user, node: nil, endpoint_id: nil}}

      _ ->
        {:error, CMS.ErrorCat.mentions_map_not_found()}
    end
  end

  defp resolve_list_root(community, direction, %{type: type} = input, budget)
       when type in [:post, :blog, :changelog, :doc, :ticket, :comment] do
    locator = Map.get(input, :locator)

    with true <- is_map(locator) and Map.get(locator, :type) == type,
         {:ok, [node]} <- NodeStatus.resolve_locators(community, [locator], budget),
         {:ok, :pass} <- validate_focus(node) do
      case direction do
        :outgoing ->
          {:ok, %{identity: node.identity, node: node, endpoint_id: node.endpoint_id, user: nil}}

        :incoming ->
          {:ok, %{identity: node.identity, node: node, endpoint_id: node.endpoint_id, user: nil}}
      end
    else
      false -> {:error, CMS.ErrorCat.mentions_map_invalid_locator()}
      {:error, _reason} = error -> error
      _ -> {:error, CMS.ErrorCat.mentions_map_not_found()}
    end
  end

  defp resolve_list_root(_community, _direction, _input, _budget),
    do: {:error, CMS.ErrorCat.mentions_map_invalid_locator()}

  defp list_edge_page(
         _community,
         :outgoing,
         %{node: %{sync_state: :pending}},
         _revision,
         page_request,
         _budget
       ) do
    {:ok, [], empty_list_page(page_request)}
  end

  defp list_edge_page(community, direction, root, revision, page_request, budget) do
    query =
      from(edge in CurrentEdge,
        where: edge.source_endpoint_id in subquery(visible_current_endpoint_ids(community)),
        order_by: [asc: edge.edge_key]
      )
      |> constrain_list_root(direction, root)

    case page_request do
      %Params.Offset{page: page, size: size} ->
        total =
          Budget.run!(budget, fn timeout ->
            Repo.aggregate(query, :count, :id, timeout: timeout)
          end)

        offset = (page - 1) * size

        rows =
          query
          |> limit(^size)
          |> offset(^offset)
          |> then(&Budget.run!(budget, fn timeout -> Repo.all(&1, timeout: timeout) end))

        page_info = %Page.Offset{
          entries: rows,
          page: page,
          size: size,
          total_count: total,
          total_pages: if(total == 0, do: 0, else: ceil(total / size)),
          has_previous_page: page > 1,
          has_next_page: page * size < total
        }

        {:ok, rows, page_info}

      %Params.Cursor{first: first, after: cursor} ->
        with {:ok, after_key} <-
               decode_list_cursor(cursor, community.id, revision, direction, root.identity) do
          rows =
            query
            |> maybe_after_edge(after_key)
            |> limit(^(first + 1))
            |> then(&Budget.run!(budget, fn timeout -> Repo.all(&1, timeout: timeout) end))

          has_next_page = length(rows) > first
          selected = Enum.take(rows, first)
          last_key = if has_next_page, do: List.last(selected).edge_key

          with {:ok, end_cursor} <-
                 encode_list_cursor(community.id, revision, direction, root.identity, last_key) do
            {:ok, selected,
             %Page.Cursor{
               entries: selected,
               start_cursor: nil,
               end_cursor: end_cursor,
               has_previous_page: not is_nil(cursor),
               has_next_page: has_next_page
             }}
          end
        end
    end
  end

  defp empty_list_page(%Params.Offset{page: page, size: size}) do
    %Page.Offset{
      entries: [],
      page: page,
      size: size,
      total_count: 0,
      total_pages: 0,
      has_previous_page: page > 1,
      has_next_page: false
    }
  end

  defp empty_list_page(%Params.Cursor{after: cursor}) do
    %Page.Cursor{
      entries: [],
      start_cursor: nil,
      end_cursor: cursor,
      has_previous_page: not is_nil(cursor),
      has_next_page: false
    }
  end

  defp constrain_list_root(query, :outgoing, %{endpoint_id: endpoint_id}) do
    where(query, [edge], edge.source_endpoint_id == ^endpoint_id)
  end

  defp constrain_list_root(query, :incoming, %{endpoint_id: endpoint_id})
       when is_binary(endpoint_id) do
    where(query, [edge], edge.target_endpoint_id == ^endpoint_id)
  end

  defp constrain_list_root(query, :incoming, %{identity: {:user, user_id}}) do
    where(query, [edge], edge.target_type == :user and edge.target_user_id == ^user_id)
  end

  defp constrain_list_root(query, _direction, _root), do: where(query, [edge], false)

  defp maybe_after_edge(query, nil), do: query
  defp maybe_after_edge(query, edge_key), do: where(query, [edge], edge.edge_key > ^edge_key)

  defp project_list_entries(
         _community,
         :outgoing,
         %{node: %{sync_state: :pending}},
         _rows,
         _budget
       ),
       do: {:ok, []}

  defp project_list_entries(community, direction, root, rows, budget) do
    endpoint_ids =
      rows
      |> Enum.flat_map(&[&1.source_endpoint_id, &1.target_endpoint_id])
      |> Enum.reject(&is_nil/1)
      |> Enum.uniq()

    endpoints =
      case endpoint_ids do
        [] ->
          []

        ids ->
          Budget.run!(budget, fn timeout ->
            Repo.all(from(endpoint in Endpoint, where: endpoint.id in ^ids), timeout: timeout)
          end)
      end

    with {:ok, nodes_by_identity} <- NodeStatus.load_endpoints(community, endpoints, budget),
         users <- load_list_users(rows, root, budget) do
      nodes_by_endpoint =
        nodes_by_identity
        |> Map.values()
        |> Map.new(&{&1.endpoint_id, &1})

      Enum.reduce_while(rows, {:ok, []}, fn edge, {:ok, acc} ->
        source =
          if direction == :outgoing,
            do: root.node,
            else: Map.get(nodes_by_endpoint, edge.source_endpoint_id)

        target =
          case edge.target_type do
            :url -> %{__typename: :mention_url_target, type: :url, url: edge.normalized_url}
            :user -> user_target(Map.get(users, edge.target_user_id))
            _ when direction == :outgoing -> Map.get(nodes_by_endpoint, edge.target_endpoint_id)
            _ -> root.node
          end

        if is_nil(source) or is_nil(target) or source.status != :readable do
          {:halt, {:error, CMS.ErrorCat.mentions_map_invalid_request()}}
        else
          entry = %{
            id: Opaque.id(:edge, community.id, edge.edge_key),
            source: source,
            target_type: edge.target_type,
            target: target,
            mention_case: edge.mention_case,
            occurrences_count: occurrence_count(edge.occurrences),
            mentioned_at: edge.mentioned_at
          }

          {:cont, {:ok, [entry | acc]}}
        end
      end)
      |> case do
        {:ok, entries} -> {:ok, Enum.reverse(entries)}
        {:error, _reason} = error -> error
      end
    end
  end

  defp load_list_users(rows, root, budget) do
    ids =
      rows
      |> Enum.filter(&(&1.target_type == :user))
      |> Enum.map(& &1.target_user_id)
      |> Kernel.++(if(root.identity |> elem(0) == :user, do: [elem(root.identity, 1)], else: []))
      |> Enum.reject(&is_nil/1)
      |> Enum.uniq()

    case ids do
      [] ->
        %{}

      _ ->
        Budget.run!(budget, fn timeout ->
          Repo.all(from(user in User, where: user.id in ^ids, select: user), timeout: timeout)
        end)
        |> Map.new(&{&1.id, &1})
    end
  end

  defp user_target(%User{} = user) do
    login = user.login

    %{
      __typename: :mention_user_target,
      type: :user,
      status: :readable,
      login: login,
      display_name: user.nickname || login,
      url: "/u/#{URI.encode(login)}"
    }
  end

  defp user_target(_),
    do: %{
      __typename: :mention_user_target,
      type: :user,
      status: :unavailable,
      login: nil,
      display_name: nil,
      url: nil
    }

  defp occurrence_count(%{"items" => items}) when is_list(items), do: length(items)
  defp occurrence_count(%{items: items}) when is_list(items), do: length(items)
  defp occurrence_count(_), do: 0

  defp encode_list_cursor(_community_id, _revision, _direction, _identity, nil), do: {:ok, nil}

  defp encode_list_cursor(community_id, revision, direction, identity, edge_key) do
    key = Application.fetch_env!(:groupher_server, :mentions_map_key)

    payload =
      :erlang.term_to_binary(
        {Opaque.version(), :one_hop, community_id, revision, direction, identity, edge_key}
      )

    signature = :crypto.mac(:hmac, :sha256, key, payload)
    {:ok, Base.url_encode64(payload <> signature, padding: false)}
  end

  defp decode_list_cursor(nil, _community_id, _revision, _direction, _identity), do: {:ok, nil}

  defp decode_list_cursor(cursor, community_id, revision, direction, identity)
       when is_binary(cursor) and byte_size(cursor) <= 4096 do
    key = Application.fetch_env!(:groupher_server, :mentions_map_key)

    with {:ok, decoded} <- Base.url_decode64(cursor, padding: false),
         true <- byte_size(decoded) > 32,
         payload_size <- byte_size(decoded) - 32,
         <<payload::binary-size(^payload_size), signature::binary-size(32)>> <- decoded,
         expected <- :crypto.mac(:hmac, :sha256, key, payload),
         true <- Plug.Crypto.secure_compare(signature, expected),
         {version, :one_hop, ^community_id, ^revision, ^direction, ^identity, edge_key} <-
           :erlang.binary_to_term(payload, [:safe]),
         true <- version == Opaque.version() and is_binary(edge_key) do
      {:ok, edge_key}
    else
      _ -> {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}
    end
  rescue
    _ -> {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}
  end

  defp decode_list_cursor(_, _, _, _, _),
    do: {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}

  defp load_frontier_edges(community, revision, nodes, inputs, first, budget) do
    Enum.zip(nodes, inputs)
    |> Enum.reduce_while({:ok, []}, fn {node, input}, {:ok, pages} ->
      case frontier_page(community, revision, node, input, first, budget) do
        {:ok, page} -> {:cont, {:ok, pages ++ [page]}}
        {:error, _reason} = error -> {:halt, error}
      end
    end)
  end

  defp frontier_page(community, revision, %{can_expand: false} = node, _input, first, _budget) do
    state = if node.status == :unavailable, do: :unavailable, else: :pending

    {:ok,
     %{
       node: node,
       state: state,
       rows: [],
       raw_rows: [],
       candidate_rows: [],
       community_id: community.id,
       revision: revision,
       first: first,
       has_more?: false,
       cursor: nil
     }}
  end

  defp frontier_page(community, revision, %{endpoint_id: nil} = node, _input, first, _budget) do
    {:ok,
     %{
       node: node,
       state: :available,
       rows: [],
       raw_rows: [],
       candidate_rows: [],
       community_id: community.id,
       revision: revision,
       first: first,
       has_more?: false,
       cursor: nil
     }}
  end

  defp frontier_page(community, revision, node, input, first, budget) do
    with {:ok, after_key} <-
           decode_cursor(Map.get(input, :after), community.id, revision, node.identity) do
      query =
        from(edge in CurrentEdge,
          where:
            (edge.source_endpoint_id == ^node.endpoint_id or
               edge.target_endpoint_id == ^node.endpoint_id) and
              not is_nil(edge.target_endpoint_id) and
              edge.source_endpoint_id in subquery(visible_current_endpoint_ids(community)),
          order_by: [asc: edge.edge_key],
          limit: ^(first + 1)
        )

      query =
        if is_nil(after_key),
          do: query,
          else: where(query, [edge], edge.edge_key > ^after_key)

      rows = Budget.run!(budget, fn timeout -> Repo.all(query, timeout: timeout) end)

      {:ok,
       %{
         node: node,
         state: :available,
         rows: [],
         raw_rows: rows,
         candidate_rows: [],
         community_id: community.id,
         revision: revision,
         first: first,
         has_more?: false,
         cursor: nil
       }}
    end
  end

  defp load_edge_node_statuses(community, edge_pages, budget) do
    endpoint_ids =
      edge_pages
      |> Enum.flat_map(& &1.raw_rows)
      |> Enum.flat_map(&[&1.source_endpoint_id, &1.target_endpoint_id])
      |> Enum.uniq()

    endpoints =
      case endpoint_ids do
        [] ->
          []

        _ ->
          Budget.run!(budget, fn timeout ->
            Repo.all(from(endpoint in Endpoint, where: endpoint.id in ^endpoint_ids),
              timeout: timeout
            )
          end)
      end

    with {:ok, nodes} <- NodeStatus.load_endpoints(community, endpoints, budget) do
      {:ok, nodes}
    end
  end

  defp attach_node_statuses(
         community,
         revision,
         first,
         edge_pages,
         frontier_nodes,
         nodes_by_identity
       ) do
    node_by_endpoint =
      nodes_by_identity
      |> Map.values()
      |> Enum.reject(&is_nil(&1.endpoint_id))
      |> Map.new(&{&1.endpoint_id, &1})

    frontier_by_endpoint =
      frontier_nodes
      |> Enum.filter(& &1.endpoint_id)
      |> Map.new(&{&1.endpoint_id, &1})

    pages =
      Enum.map(edge_pages, fn page ->
        candidate_rows =
          Enum.flat_map(page.raw_rows, fn row ->
            source =
              Map.get(frontier_by_endpoint, row.source_endpoint_id) ||
                Map.get(node_by_endpoint, row.source_endpoint_id)

            target =
              Map.get(frontier_by_endpoint, row.target_endpoint_id) ||
                Map.get(node_by_endpoint, row.target_endpoint_id)

            cond do
              is_nil(source) or is_nil(target) ->
                []

              page.node.endpoint_id == row.target_endpoint_id and source.status != :readable ->
                []

              true ->
                [%{row: row, source_node: source, target_node: target}]
            end
          end)

        %{page | candidate_rows: candidate_rows}
      end)

    Enum.reduce_while(pages, {:ok, []}, fn page, {:ok, acc} ->
      candidates =
        Enum.map(page.candidate_rows, fn item ->
          Map.merge(item.row, Map.take(item, [:source_node, :target_node]))
        end)

      cond do
        true ->
          has_more? = length(candidates) > first
          selected = Enum.take(candidates, first)

          cursor =
            if has_more? and selected != [] do
              encode_cursor(
                community.id,
                revision,
                page.node.identity,
                List.last(selected).edge_key
              )
            end

          complete = %{
            page
            | rows: selected,
              has_more?: has_more?,
              cursor: cursor,
              community_id: community.id,
              revision: revision,
              first: first
          }

          {:cont, {:ok, acc ++ [complete]}}
      end
    end)
  end

  defp validate_focus(%{status: :readable}), do: {:ok, :pass}
  defp validate_focus(_), do: {:error, CMS.ErrorCat.mentions_map_not_found()}

  defp validate_frontiers(focus_node, nodes) do
    identities = Enum.map(nodes, & &1.identity)

    frequencies = Enum.frequencies(identities)

    duplicate_frontiers =
      Enum.any?(frequencies, fn
        {identity, count} when identity == focus_node.identity -> count > 1
        {_identity, count} -> count > 1
      end)

    cond do
      Enum.any?(nodes, &is_nil(&1.identity)) ->
        if Enum.any?(nodes, &(&1.status == :unavailable)),
          do: {:ok, :pass},
          else: {:error, CMS.ErrorCat.mentions_map_invalid_locator()}

      duplicate_frontiers ->
        {:error, CMS.ErrorCat.mentions_map_duplicate_frontier()}

      true ->
        {:ok, :pass}
    end
  end

  defp projection_pending(community, _nodes, budget),
    do: {:ok, public_projection_pending?(community, budget)}

  defp public_projection_pending?(community, budget) do
    public_sources =
      [:post, :blog, :changelog]
      |> Enum.map(&pending_article_sources(&1, community))
      |> Enum.concat([
        pending_doc_sources(community),
        pending_ticket_sources(community),
        pending_comment_sources(community)
      ])
      |> Enum.reduce(fn query, acc -> union_all(acc, ^query) end)
      |> subquery()

    Budget.run!(budget, fn timeout ->
      Repo.exists?(from(source in subquery(public_sources), select: 1, limit: 1),
        timeout: timeout
      )
    end)
  end

  defp validate_article_comment_focus(%{status: :readable, identity: {type, _, nil, _}})
       when type in [:post, :blog, :changelog, :doc],
       do: {:ok, :pass}

  defp validate_article_comment_focus(_), do: {:error, CMS.ErrorCat.mentions_map_not_found()}

  defp parse_comment_cursor(nil), do: {:ok, 0}

  defp parse_comment_cursor(value) when is_binary(value) do
    case Integer.parse(value) do

  defp comment_seed_rows(community, article_id, after_id, limit, budget) do
    query =
      CMS.Gate.scope(Comment, nil, :read, CommentScope.all_public())
      |> scope_query()
      |> where([comment, ...], as(:gate_comment_lifecycle).state == :visible)
      |> join(:inner, [comment, ...], parent in subquery(public_comment_parents(community)),
        on: parent.article_id == comment.article_id
      )
      |> where(
        [comment, ...],
        comment.article_id == ^article_id and comment.inner_id > ^after_id and
          as(:gate_community).id == ^community.id
      )

    query
    |> join(:left, [comment, ...], parent_endpoint in Endpoint,
      on:
        parent_endpoint.article_id == comment.article_id and
          parent_endpoint.type == comment.thread,
      as: :mention_parent_endpoint
    )
    |> join(:left, [comment, ...], endpoint in Endpoint,
      on:
        endpoint.type == :comment and endpoint.comment_id == comment.id and
          endpoint.parent_endpoint_id == as(:mention_parent_endpoint).id,
      as: :mention_endpoint
    )
    |> join(:left, [comment, ...], state in ProjectionState,
      on: state.source_endpoint_id == as(:mention_endpoint).id,
      as: :mention_projection_state
    )
    |> join(:left, [comment, ...], edge in CurrentEdge,
      on: edge.source_endpoint_id == as(:mention_endpoint).id,
      as: :mention_current_edge
    )
    |> where(
      [comment, ...],
      not is_nil(as(:mention_current_edge).id) or is_nil(as(:mention_endpoint).id) or
        is_nil(as(:mention_projection_state).source_endpoint_id) or
        is_nil(as(:mention_projection_state).synced_set_id) or
        is_nil(as(:mention_projection_state).synced_content_token) or
        as(:mention_projection_state).desired_set_id !=
          as(:mention_projection_state).synced_set_id or
        as(:mention_projection_state).desired_content_token !=
          as(:mention_projection_state).synced_content_token
    )
    |> select([comment, ...], %{
      article_inner_id: as(:gate_comment_binding).inner_id,
      comment_inner_id: comment.inner_id
    })
    |> distinct(true)
    |> order_by([comment, ...], asc: comment.inner_id)
    |> limit(^limit)
    |> then(&Budget.run!(budget, fn timeout -> Repo.all(&1, timeout: timeout) end))
  end

  defp visible_current_endpoint_ids(community) do
    ordinary =
      [:post, :blog, :changelog]
      |> Enum.map(fn type ->
        scope =
          CMS.Gate.scope(Article, nil, :read, ArticleScope.public(type)) |> scope_query()

        scope
        |> join(:inner, [article, ...], endpoint in Endpoint,
          on: endpoint.type == ^type and endpoint.article_id == article.id,
          as: :mention_endpoint
        )
        |> where([article, endpoint, ...], as(:gate_community).id == ^community.id)
        |> current_endpoint_selection(:article)
      end)

    doc =
      CMS.Gate.scope(Article, nil, :read, DocScope.public_main())
      |> scope_query()
      |> join(:inner, [article, ...], endpoint in Endpoint,
        on: endpoint.type == :doc and endpoint.article_id == article.id,
        as: :mention_endpoint
      )
      |> where([article, endpoint, ...], as(:gate_community).id == ^community.id)
      |> current_endpoint_selection(:doc)

    ticket =
      CMS.Gate.scope(TicketState, nil, :read, TicketScope.public(community))
      |> scope_query()
      |> join(:inner, [ticket_state: state], endpoint in Endpoint,
        on: endpoint.type == :ticket and endpoint.article_id == state.article_id,
        as: :mention_endpoint
      )
      |> current_endpoint_selection(:ticket)

    comments =
      CMS.Gate.scope(Comment, nil, :read, CommentScope.all_public())
      |> scope_query()
      |> join(:inner, [comment, ...], parent in subquery(public_comment_parents(community)),
        on: parent.article_id == comment.article_id
      )
      |> where([comment, ...], as(:gate_community).id == ^community.id)
      |> join(:inner, [comment, ...], parent_endpoint in Endpoint,
        on:
          parent_endpoint.article_id == comment.article_id and
            parent_endpoint.type == comment.thread,
        as: :mention_parent_endpoint
      )
      |> join(:inner, [comment, ...], endpoint in Endpoint,
        on:
          endpoint.type == :comment and endpoint.comment_id == comment.id and
            endpoint.parent_endpoint_id == as(:mention_parent_endpoint).id,
        as: :mention_endpoint
      )
      |> join(:inner, [comment, ...], body in CommentBody,
        on: body.comment_id == comment.id,
        as: :mention_comment_body
      )
      |> current_endpoint_selection(:comment)

    [ordinary, doc, ticket, comments]
    |> List.flatten()
    |> Enum.reject(&is_nil/1)
    |> union_queries()
  end

  defp public_comment_parents(community) do
    ordinary =
      [:post, :blog, :changelog]
      |> Enum.map(fn type ->
        CMS.Gate.scope(Article, nil, :read, ArticleScope.public(type))
        |> scope_query()
        |> where([article, ...], as(:gate_community).id == ^community.id)
        |> select([article, ...], %{
          article_id: article.id
        })
      end)

    doc =
      CMS.Gate.scope(Article, nil, :read, DocScope.public_main())
      |> scope_query()
      |> where([article, ...], as(:gate_community).id == ^community.id)
      |> select([article, ...], %{
        article_id: article.id
      })

    Enum.reduce(ordinary ++ [doc], fn query, acc -> union_all(acc, ^query) end)
  end

  defp current_endpoint_selection(query, source_type) do
    query
    |> join(:inner, [...], state in ProjectionState,
      on: state.source_endpoint_id == as(:mention_endpoint).id,
      as: :mention_projection_state
    )
    |> join(:inner, [...], set in MentionSet,
      on:
        set.id == as(:mention_projection_state).synced_set_id and
          set.source_endpoint_id == as(:mention_endpoint).id and
          as(:mention_projection_state).desired_set_id == set.id and
          as(:mention_projection_state).synced_set_id == set.id and
          as(:mention_projection_state).desired_content_token ==
            as(:mention_projection_state).synced_content_token and set.parser_version == 1
    )
    |> current_canonical_token(source_type)
    |> select([...], %{id: as(:mention_endpoint).id})
  end

  defp current_canonical_token(query, :article) do
    where(
      query,
      [article, ..., mention_projection_state: state],
      state.desired_content_token ==
        fragment(
          "concat(?, ':', ?)",
          as(:gate_article_public).revision_id,
          as(:gate_article_public).publication_version
        )
    )
  end

  defp current_canonical_token(query, :doc) do
    where(
      query,
      [article, ..., mention_projection_state: state],
      state.desired_content_token ==
        fragment(
          "concat(?, ':', ?, ':', ?)",
          as(:gate_doc_branch).id,
          as(:gate_article_public).branch_version_id,
          as(:gate_article_public).publication_version
        )
    )
  end

  defp current_canonical_token(query, :ticket) do
    where(
      query,
      [..., mention_projection_state: state],
      state.desired_content_token ==
        fragment(
          "concat(?, ':', ?)",
          as(:ticket_public).revision_id,
          as(:ticket_public).publication_version
        )
    )
  end

  defp current_canonical_token(query, :comment) do
    where(
      query,
      [comment, ..., mention_projection_state: state],
      state.desired_content_token ==
        fragment(
          "concat(?, ':', ?, ':', ?)",
          comment.id,
          as(:mention_comment_body).version,
          as(:mention_comment_body).body_hash
        )
    )
  end

  defp union_queries([query]), do: query

  defp union_queries([query | rest]) do
    Enum.reduce(rest, query, fn next, acc -> union_all(acc, ^next) end)
  end

  defp pending_comment_sources(community) do
    parent_sources =
      [:post, :blog, :changelog]
      |> Enum.map(fn type ->
        scope = CMS.Gate.scope(Article, nil, :read, ArticleScope.public(type)) |> scope_query()

        scope
        |> where([article, ...], as(:gate_community).id == ^community.id)
        |> select([article, ...], %{
          article_id: article.id
        })
      end)
      |> Enum.concat([pending_doc_parent_sources(community)])
      |> Enum.reduce(fn query, acc -> union_all(acc, ^query) end)
      |> subquery()

    comment_scope =
      CMS.Gate.scope(Comment, nil, :read, CommentScope.all_public())
      |> scope_query()
      |> where([comment, ...], as(:gate_comment_lifecycle).state == :visible)

    comment_scope
    |> join(:inner, [comment, ...], parent in subquery(parent_sources),
      on: comment.article_id == parent.article_id
    )
    |> where([comment, ...], as(:gate_community).id == ^community.id)
    |> select([comment, ...], %{comment_id: comment.id})
    |> subquery()
    |> pending_endpoint_source(:comment, :comment_id)
  end

  defp pending_endpoint_source(source, type, identity_field) do
    from(identity in subquery(source),
      left_join: endpoint in Endpoint,
      on:
        endpoint.type == ^type and
          field(endpoint, ^identity_field) == field(identity, ^identity_field),
      left_join: state in ProjectionState,
      on: state.source_endpoint_id == endpoint.id,
      left_join: set in MentionSet,
      on: set.id == state.desired_set_id and set.source_endpoint_id == endpoint.id,
      where:
        is_nil(endpoint.id) or is_nil(state.source_endpoint_id) or
          is_nil(state.synced_set_id) or is_nil(state.synced_content_token) or
          state.desired_content_token != state.synced_content_token or
          is_nil(set.id) or set.parser_version != 1,
      select: %{pending: type(^true, :boolean)}
    )
  end

  defp unchanged_revision(revision, budget) do
    case Revision.current(budget) do
      {:ok, ^revision} -> {:ok, :pass}
      {:ok, _changed} -> {:error, CMS.ErrorCat.mentions_map_invalid_request()}
      {:error, _reason} = error -> error
    end
  end

  defp community(slug, budget) do
    case Budget.run!(budget, fn timeout ->
           Repo.get_by(Community, [slug: slug], timeout: timeout)
         end) do
      %Community{} = community -> {:ok, community}
      _ -> {:error, CMS.ErrorCat.mentions_map_not_found()}
    end
  end

  defp transaction(budget, fun) do
    scoped_fun = fn ->
      with {:ok, :pass} <- Budget.arm_statement_timeout!(budget) do
        fun.()
      end
    end

    if Repo.in_transaction?() do
      scoped_fun.()
    else
      case Repo.transaction(scoped_fun) do
        {:ok, result} -> result
        {:error, reason} -> {:error, reason}
      end
    end
  end

  defp encode_cursor(community_id, revision, identity, edge_key) do
    key = Application.fetch_env!(:groupher_server, :mentions_map_key)

    payload =
      :erlang.term_to_binary({Opaque.version(), community_id, revision, identity, edge_key})

    signature = :crypto.mac(:hmac, :sha256, key, payload)
    Base.url_encode64(payload <> signature, padding: false)
  end

  defp decode_cursor(nil, _community_id, _revision, _identity), do: {:ok, nil}

  defp decode_cursor(cursor, community_id, revision, identity)
       when is_binary(cursor) and byte_size(cursor) <= 4096 do
    key = Application.fetch_env!(:groupher_server, :mentions_map_key)

    with {:ok, decoded} <- Base.url_decode64(cursor, padding: false),
         true <- byte_size(decoded) > 32,
         payload_size <- byte_size(decoded) - 32,
         <<payload::binary-size(^payload_size), signature::binary-size(32)>> <- decoded,
         expected <- :crypto.mac(:hmac, :sha256, key, payload),
         true <- Plug.Crypto.secure_compare(signature, expected),
         {version, ^community_id, ^revision, ^identity, edge_key} <-
           :erlang.binary_to_term(payload, [:safe]),
         true <- version == Opaque.version(),
         true <- is_binary(edge_key) do
      {:ok, edge_key}
    else
      _ -> {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}
    end
  rescue
    _ -> {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}
  end

  defp decode_cursor(_, _, _, _), do: {:error, CMS.ErrorCat.mentions_map_invalid_cursor()}
end
