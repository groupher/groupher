defmodule GroupherServer.CMS.ArtimentMentions.Effects do
  @moduledoc """
  Records durable Mention projection intents inside their source transaction.

      saved immutable set -> Outbox event -> projector worker
  """

  alias GroupherServer.CMS
  @doc "Enqueues one set-scoped projection event through the CMS Outbox."
  @spec enqueue_projection(Ecto.UUID.t(), Ecto.UUID.t(), tuple(), [integer()]) ::
          {:ok, CMS.Outbox.Event.t()} | {:error, term()}
  def enqueue_projection(source_endpoint_id, set_id, identity, recipient_ids \\ [])

  def enqueue_projection(source_endpoint_id, set_id, identity, recipient_ids)
      when is_binary(source_endpoint_id) and is_binary(set_id) and is_tuple(identity) do
    CMS.Outbox.send(%{
      event: "mention.set_project",
      resource_type: "mention_source",
      resource_id: source_endpoint_id,
      identity: identity,
      effect_key: "mention.set_project:#{source_endpoint_id}:#{set_id}",
      worker: GroupherServer.Jobs.CMSMentionSetProjector,
      data: %{
        source_endpoint_id: source_endpoint_id,
        set_id: set_id,
        notify_user_ids: recipient_ids
      }
    })
  end

  def enqueue_projection(_, _, _, _), do: {:error, :invalid_mention_projection_event}
end
